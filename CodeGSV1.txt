function sanitizeInput(input) {
  if (typeof input !== 'string') return input;
  
  // Secures against script tags but leaves slashes (/), ampersands (&), and quotes (' ") alone
  // so your Google Sheet remains perfectly readable and invoice matching works!
  return input
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .trim();
}

function isMonthLocked(dateString) {
  if (!dateString) return false;
  try {
    const d = new Date(dateString);
    const mKey = d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, '0');
    const sheet = getGSTLockSheet();
    const data = sheet.getDataRange().getDisplayValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === mKey && data[i][1] === "TRUE") {
        return true;
      }
    }
  } catch (e) { console.error("Lock Check Error", e); }
  return false;
}

// PREMIUM OPTION A: Dynamic Financial Year logic
function getCurrentFY() {
  const d = new Date();
  const y = d.getFullYear();
  const m = d.getMonth() + 1;
  if (m >= 4) return `${String(y).slice(-2)}-${String(y + 1).slice(-2)}`;
  return `${String(y - 1).slice(-2)}-${String(y).slice(-2)}`;
}

function isFrozen(dateStr) {
  if (!dateStr) return false;
  const conf = getConfig();
  const freezeDays = parseInt(conf.FREEZE_DAYS);
  if (isNaN(freezeDays) || freezeDays <= 0) return false;

  const recDate = new Date(dateStr);
  if (isNaN(recDate.getTime())) return false; // ADDED: Failsafe for invalid dates

  const today = new Date();
  const diffDays = Math.floor((today - recDate) / (1000 * 60 * 60 * 24));
  return diffDays > freezeDays;
}

// PREMIUM: Unified Period Lock Gatekeeper
function checkPeriodLocks(dateStr) {
  if (!dateStr) return;
  if (isFrozen(dateStr)) {
    throw new Error("Transaction Freeze Active: Cannot add or edit records older than your configured freeze days.");
  }
  if (isMonthLocked(dateStr)) {
    throw new Error("Period Locked: This financial month has been locked for GST compliance. Unlock via Admin to edit.");
  }
}

function hardValidate(d, type) {
  if (d.date) {
    let recDate = new Date(d.date);
    recDate.setHours(0, 0, 0, 0);
    let today = new Date();
    today.setHours(0, 0, 0, 0);
    if (recDate > today) throw new Error("Validation Error: Future dates are strictly restricted.");
    let fyYear = recDate.getFullYear();
    if (fyYear < 2000 || fyYear > 2100) throw new Error("Validation Error: Invalid Financial Year.");
  }
}

function manageJournalEntry(sourceType, sourceId, date, entries) {
  // ENTERPRISE FIX: Block Unbalanced Entries
  let totalDr = 0, totalCr = 0;
  entries.forEach(e => {
    totalDr += (parseFloat(e.dr) || 0);
    totalCr += (parseFloat(e.cr) || 0);
  });
  if (Math.abs(totalDr - totalCr) > 0.01) {
    throw new Error(`Financial Integrity Block: Journal is unbalanced! (Dr: ₹${totalDr.toFixed(2)}, Cr: ₹${totalCr.toFixed(2)})`);
  }

  checkPeriodLocks(date);

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("Journal");
  if (!sheet) {
    sheet = ss.insertSheet("Journal");
    sheet.appendRow(["PostingID", "SourceType", "SourceID", "Date", "FY", "Account", "Debit", "Credit", "IsReversed", "Timestamp"]);
  }

  // 🚀 ENTERPRISE SPEED FIX: Batch Journal Reversals in memory
  const data = sheet.getDataRange().getValues();
  let statusCol = sheet.getRange(1, 9, data.length, 1).getValues();
  let changed = false;

  for (let i = 1; i < data.length; i++) {
    if (data[i][1] === sourceType && data[i][2] === sourceId && String(statusCol[i][0]).toUpperCase() !== "TRUE") {
      statusCol[i][0] = "TRUE";
      changed = true;
    }
  }
  // Write the entire column back instantly in 0.1 seconds
  if (changed) sheet.getRange(1, 9, data.length, 1).setValues(statusCol);

  const ts = new Date().toISOString();
  const fy = getCurrentFY(); // Applied Dynamic FY
  const postingId = "JRN-" + Date.now() + "-" + Math.floor(Math.random() * 1000);

  // FAST BATCH WRITING
  let rowsToAppend = [];
  entries.forEach(e => {
    if (e.dr > 0 || e.cr > 0) {
      rowsToAppend.push([postingId, sourceType, sourceId, date, fy, e.acc, e.dr, e.cr, "FALSE", ts]);
    }
  });

  if (rowsToAppend.length > 0) {
    sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend);
  }
}

function validateGSTINBackend(gstin, state) {
  if (!gstin) return { valid: true }; // Empty is allowed unless strictly enforced

  const gstinUpper = gstin.toUpperCase().trim();
  const gstRegex = /^[0-9]{2}[A-Z]{5}[0-9]{4}[A-Z]{1}[1-9A-Z]{1}Z[0-9A-Z]{1}$/;

  // 1. Format Check
  if (!gstRegex.test(gstinUpper)) {
    return { valid: false, message: "Invalid format. GSTIN must be exactly 15 characters (e.g., 36ABCDE1234F1Z5)." };
  }

  // 2. True Checksum Math Verification
  const chars = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
  let hash = 0;
  for (let i = 0; i < 14; i++) {
    let val = chars.indexOf(gstinUpper[i]);
    let prod = val * (i % 2 === 0 ? 1 : 2);
    hash += Math.floor(prod / 36) + (prod % 36);
  }
  let checkDigit = (36 - (hash % 36)) % 36;

  if (chars[checkDigit] !== gstinUpper[14]) {
    return { valid: false, message: "Checksum Failed.The GSTIN is mathematically incorrect." };
  }

  // 3. State code verification (Exact Match)
  const stateCode = gstinUpper.substring(0, 2);
  const stateMap = {
    "01": "Jammu", "02": "Himachal Pradesh", "03": "Punjab", "04": "Chandigarh",
    "05": "Uttarakhand", "06": "Haryana", "07": "Delhi", "08": "Rajasthan", "09": "Uttar Pradesh",
    "10": "Bihar", "11": "Sikkim", "12": "Arunachal Pradesh", "13": "Nagaland", "14": "Manipur",
    "15": "Mizoram", "16": "Tripura", "17": "Meghalaya", "18": "Assam", "19": "West Bengal",
    "20": "Jharkhand", "21": "Odisha", "22": "Chhattisgarh", "23": "Madhya Pradesh", "24": "Gujarat",
    "25": "Daman and Diu", "26": "Dadra and Nagar Haveli", "27": "Maharashtra", "29": "Karnataka",
    "30": "Goa", "31": "Lakshadweep", "32": "Kerala", "33": "Tamil Nadu", "34": "Puducherry",
    "35": "Andaman and Nicobar", "36": "Telangana", "37": "Andhra Pradesh", "38": "Ladakh","06": "Haryana"
  };

  if (state && stateMap[stateCode]) {
    if (stateMap[stateCode].toLowerCase() !== state.toLowerCase()) {
      return { valid: false, message: `State Mismatch! GSTIN starts with '${stateCode}' (${stateMap[stateCode]}), but you selected '${state}'.` };
    }
  }

  return { valid: true };
}

// ==========================================
function validateSession(token, requiredRole = "user") {
  if (!token) throw new Error("Security Block: No session token provided.");

  const cleanToken = String(token).replace(/['"]/g, '').trim();
  const sessionKey = "SESSION_" + cleanToken;

  // 1️⃣ Try the Cache first (Super Fast)
  let sessionRole = CacheService.getScriptCache().get(sessionKey);

  // 2️⃣ Fallback to Properties (Slower, but 100% Reliable)
  if (!sessionRole) {
    sessionRole = PropertiesService.getScriptProperties().getProperty(sessionKey);

    // Optional: If found in properties, put it back in cache for speed next time!
    if (sessionRole) {
      CacheService.getScriptCache().put(sessionKey, sessionRole, 21600);
    }
  }

  if (!sessionRole) {
    throw new Error("Security Block: Session expired. Please log in again.");
  }

  // 🛑 RBAC ENFORCEMENT
  if (requiredRole === "admin" && sessionRole !== "admin") {
    throw new Error("Access Denied: Administrator privileges required.");
  }

  return sessionRole;
}

      }

      else if (type === "Tax Invoice") {
        if (paid >= (total - 1)) status = "Received"; // Marks as Received if Advance was adjusted!
        else if (paid > 0) status = "Partial";
        else status = "Pending";
      }

      if (type === "Tax Invoice" && !isCancelled) {
        realInvoiceCount++;
        if (status === "Received") countReceived++;
        else countPending++;

        sales += total;
        received += paid;
        const due = total - paid;

        let dObj = new Date(r[7]);
        if (!isNaN(dObj.getTime())) {
          let m = dObj.getFullYear() + "-" + String(dObj.getMonth() + 1).padStart(2, '0');

        stateSales[r[3] || "Unknown"] = (stateSales[r[3] || "Unknown"] || 0) + total;

        if (status !== "Received") {
          const d = new Date(r[7]);
          if (!isNaN(d.getTime())) {
            let overdueDays = parseInt(config.OVERDUE_DAYS) || 45;
            let dueDate = new Date(d.getTime() + (overdueDays * 86400000));
            let diff = Math.ceil((new Date() - dueDate) / 86400000);
            if (diff > 0) {
              if (diff <= 30) aging["0-30"] += due;
              else if (diff <= 60) aging["31-60"] += due;
              else if (diff <= 90) aging["61-90"] += due;
              else aging["90+"] += due;
            }
          }
        }
      }

      return {
        invNum: r[6], date: r[7], company: r[5], total: total, status: status, type: type,
        pdf: r[31], items: r[32], due: total - paid, siteName: r[4] || "Main Site",
        billFrom: r[8], billTo: r[9], soNumber: r[10], vehicle: r[14], revNo: r[15] || "", ship: r[16], work: r[17], terms: termsVal,
        gstin: r[12], idx: i, words: r[30],
        igst: igst, cgst: cgst, sgst: sgst, taxable: taxable, state: r[3],

      };
    }).filter(i => i);

    const nextIds = {
      SD: `SD/${currentFY}/${String(maxSD + 1).padStart(3, '0')}`,
      PI: `PI/${currentFY}/${String(maxPI + 1).padStart(3, '0')}`,

      PO: `PO/${currentFY}/${String(maxPO + 1).padStart(3, '0')}`
    };

      if (!r[0] || r[11] === "TRUE") return null;
      let amt = parseFloat(r[5]); if (isNaN(amt)) amt = 0;
      return {
        id: r[0], date: r[1], client: r[2], type: r[3], ref: r[4], amount: amt, mode: r[6], site: r[7] || "", head: r[8] || "", idx: i,
        lastModified: r[10] || "", receiptUrl: r[12] || ""
      };
    }).filter(p => p);

    const gstr2a = rawGSTR.map(r => ({
      month: r[0], gstin: r[1], name: r[2], invNo: r[3], date: r[4], taxable: parseFloat(r[5] || 0), tax: parseFloat(r[6] || 0)
    })).filter(r => r.month);

    const purchases = rawPurchases.map((r, i) => {
      if (r[11] === "TRUE") return null;
      return {
        id: r[0], date: r[1], vendor: r[2], gstin: r[3], invNo: r[4], 
        taxable: parseFloat(r[5] || 0), tax: parseFloat(r[6] || 0), total: parseFloat(r[7] || 0), 
        desc: r[8], costHead: r[9] || "", lastModified: r[10] || "", 
        status2A: r[12] || "Pending", woNo: r[13] || "", idx: i // 🚀 NEW woNo Mapping
      };
    }).filter(p => p).reverse();

    // ==========================================
    // CFO INTELLIGENCE ENGINE 
    // ==========================================
    let siteStats = {};
    let clientRev = {};
    let totalCashIn180Days = 0;
    const nowMs = Date.now();
    const day180Ms = 180 * 24 * 60 * 60 * 1000;

    invoices.forEach(inv => {
      if (inv.status !== 'Cancelled') {
        const site = inv.siteName || "Main Site";
        const comp = inv.company || "Unknown Client";
        const invTotal = parseFloat(inv.total) || 0;

        // CFO STRICT MATH: Only count actual finalized Tax Invoices (Ignore Quotations, Proformas, and Cancelled)
        if (inv.type === 'Tax Invoice' && inv.status !== 'Cancelled') {
          if (!siteStats[site]) siteStats[site] = { rev: 0, cost: 0 };
          siteStats[site].rev += invTotal;

          // PREMIUM CFO FIX: Calculate Dependency Risk based on ACTUAL UNPAID DUE, not Total Billed
          const dueAmt = parseFloat(inv.due) || 0;
          if (dueAmt > 0) {
            if (!clientRev[comp]) clientRev[comp] = 0;
            clientRev[comp] += dueAmt;
          }
        }
      }
    });

    payments.forEach(pay => {
      const rawAmt = parseFloat(pay.amount) || 0;
      const amt = Math.abs(rawAmt);
      const isOutflow = (pay.type === 'Vendor Payment' || pay.type === 'Expense' || pay.type === 'GST Payment' || rawAmt < 0);

      if (isOutflow && pay.site) {
        if (!siteStats[pay.site]) siteStats[pay.site] = { rev: 0, cost: 0 };
        siteStats[pay.site].cost += amt;
      }

      if (!isOutflow) {
        let pDate = new Date(pay.date).getTime();
        if (isNaN(pDate) && pay.date) {
          const parts = String(pay.date).split('-');
          if (parts.length === 3) pDate = new Date(`${parts[2]}-${parts[1]}-${parts[0]}`).getTime();
        }
        if (!isNaN(pDate) && (nowMs - pDate <= day180Ms)) {
          totalCashIn180Days += amt;
        }
      }
    });

    const safeSales = parseFloat(sales) || 0;
    const safeReceived = parseFloat(received) || 0;
    const acp = safeSales > 0 ? ((safeSales - safeReceived) / safeSales) * 365 : 0;
    const forecast30 = totalCashIn180Days / 6;

    let topClients = Object.entries(clientRev).sort((a, b) => b[1] - a[1]).slice(0, 3);
    let top3Rev = topClients.reduce((sum, val) => sum + (parseFloat(val[1]) || 0), 0);
    const concentrationPct = safeSales > 0 ? (top3Rev / safeSales) * 100 : 0;

    const siteProfitability = Object.keys(siteStats).map(s => {
      const r = siteStats[s].rev;
      const c = siteStats[s].cost;
      let margin = r > 0 ? ((r - c) / r) * 100 : 0;
      if (r === 0 && c > 0) margin = -100;
      return { site: s, rev: r, cost: c, margin: margin };
    }).sort((a, b) => b.rev - a.rev);

    const cfoInsights = {
      acp: Math.round(acp),
      forecast30: forecast30,
      concentrationPct: concentrationPct.toFixed(1),
      topClients: topClients.map(c => ({ name: c[0], amount: c[1] || 0 })),
      sites: siteProfitability
    };

    // SECURITY FIX: Scrub PINs from the payload
    const safeConfig = { ...config };
    delete safeConfig.ACCESS_PIN;
    delete safeConfig.USER_PIN;

    const result = JSON.stringify({
      status: "success",
      data: {
        dashboard: {
          sales, received, due: sales - received, count: realInvoiceCount,
          aging, monthSales, monthGST, stateSales,
          breakdown: { pending: countPending, received: countReceived },
          gstPaidMap: gstPaidMap,
          cfo: cfoInsights
        },
        clients,
        vendors,
        nextIds,
        config: safeConfig,
        inputs: {
          sac: rawInputs.map(r => r[0]).filter(x => x && x !== ""),
          units: rawInputs.map(r => r[1]).filter(x => x && x !== ""),
          heads: rawInputs.map(r => r[2]).filter(x => x && x !== "")
        },
        gstr2a: gstr2a,
        lockedMonths: getLockedMonthsArray(),
        costHeads: costHeads,
        gstTracker: gstTracker,
        itemMaster: itemMaster,
        recurringContracts: recurringContracts,

        // 🚀 ENTERPRISE SPEED UPGRADE: Only send the 250 most recent records to the UI
        invoices: invoices.reverse().slice(0, 500),
        payments: payments.reverse().slice(0, 500),
        purchases: purchases.slice(0, 500), // Sliced to prevent lag
        workOrders: workOrders.reverse().slice(0, 500),
        purchaseOrders: purchaseOrders.reverse().slice(0, 500)
      }
    });

    putCacheChunked(cache, "appData_final_chunks", result, 300);
    return result;

  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  }
}

function runRecurringBilling() {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Recurring_Contracts");
    const invSheet = ss.getSheetByName("Invoices");
    if (!sheet || !invSheet) return JSON.stringify({ status: "error", message: "Missing sheets." });

    const data = sheet.getDataRange().getDisplayValues();
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    let processedCount = 0;

    // 🚀 1. Create arrays to hold our batched data in memory
    const newInvoicesToSave = [];
    // Pull the entire "Next Date" column into memory at once
    const nextDatesColumn = sheet.getRange(1, 11, data.length, 1).getValues();

    for (let i = 1; i < data.length; i++) {
      if (data[i][11] === "TRUE" && data[i][10]) {
        const nextDate = new Date(data[i][10]);
        nextDate.setHours(0, 0, 0, 0);

        if (nextDate <= today) {
          const newInvNum = `DRAFT-REC-${Date.now().toString().slice(-6)}`;
          const client = data[i][1];
          const site = data[i][2];
          const amount = (parseFloat(data[i][7]) || 0) * (parseFloat(data[i][6]) || 1);
          const gst = amount * 0.18;
          const total = amount + gst;
          const itemsJson = JSON.stringify([{
            desc: data[i][3], costHead: data[i][8], sac: data[i][4], uom: data[i][5], qty: data[i][6], rate: data[i][7], amount: amount
          }]);

          const row = [
            "UID-" + Date.now() + "-" + i, "Tax Invoice", "Tax Invoice", "Andhra Pradesh", site, client, newInvNum,
            Utilities.formatDate(today, Session.getScriptTimeZone(), "yyyy-MM-dd"), "", "", "", "", "",
            "18", "", "", "", "Auto-Generated Recurring Bill", "1", "", "", "", "", "",
            gst, 0, 0, amount, total, "", "", itemsJson, "0", "", "", "DRAFT", new Date().getTime().toString(), ""
          ];

          // 🚀 2. Push to our memory array INSTEAD of appending to the sheet directly
          newInvoicesToSave.push(row);

          const freq = parseInt(data[i][9]) || 1;
          const newNextDate = new Date(nextDate);
          newNextDate.setMonth(newNextDate.getMonth() + freq);

          // 🚀 3. Update the date in our memory array INSTEAD of setting the cell directly
          nextDatesColumn[i][0] = Utilities.formatDate(newNextDate, Session.getScriptTimeZone(), "yyyy-MM-dd");

          processedCount++;
        }
      }
    }

    // 🚀 4. THE BATCH WRITE: Send everything to Google Sheets in exactly TWO commands!
    if (newInvoicesToSave.length > 0) {
      // Write all 50+ invoices at the exact same time
      invSheet.getRange(invSheet.getLastRow() + 1, 1, newInvoicesToSave.length, newInvoicesToSave[0].length).setValues(newInvoicesToSave);

      // Update all 50+ recurring dates at the exact same time
      sheet.getRange(1, 11, nextDatesColumn.length, 1).setValues(nextDatesColumn);
    }

    CacheService.getScriptCache().remove("appData_final_chunks");
    if (processedCount > 0) return JSON.stringify({ status: "success", message: `Generated drafts for ${processedCount} contracts!` });
    return JSON.stringify({ status: "success", message: "All contracts are up to date." });

  } catch (e) { return JSON.stringify({ status: "error", message: e.toString() }); }
  finally { lock.releaseLock(); }
}

function saveSettings(token, formStr) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    validateSession(token);
    const d = JSON.parse(formStr);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("Settings");
    if (!sheet) { sheet = ss.insertSheet("Settings"); sheet.appendRow(["Key", "Value"]); }

    const existingData = sheet.getDataRange().getValues();
    let configMap = new Map();
    if (existingData.length > 1) {
      for (let i = 1; i < existingData.length; i++) {
        configMap.set(String(existingData[i][0]).toUpperCase().trim(), existingData[i][1]);
      }
    }

    // --- THIS IS THE ONLY LOOP WE NEED ---
    for (let [key, value] of Object.entries(d)) {
      let dbKey = key.toUpperCase();

      // BULLETPROOF PIN PROTECTION
      if (key === 'pin' || dbKey === 'ACCESS_PIN') {
        dbKey = "ACCESS_PIN";
        let strVal = String(value).trim();
        // If the frontend sends blank or "undefined", completely SKIP overwriting the PIN
        if (!strVal || strVal === "undefined" || strVal === "null" || strVal === "") continue;
        if (strVal.length !== 64) value = hashString(strVal);
      }
      if (key === 'userPin' || dbKey === 'USER_PIN') {
        dbKey = "USER_PIN";
        let strVal = String(value).trim();
        if (!strVal || strVal === "undefined" || strVal === "null" || strVal === "") continue;
        if (strVal.length !== 64) value = hashString(strVal);
      }

      if (key === 'folder') dbKey = "DRIVE_FOLDER";
      if (key === 'logo') dbKey = "LOGO_URL";
      if (key === 'signature') dbKey = "SIGNATURE_URL";
      if (key === 'upi') dbKey = "UPI_ID";
      if (key === 'overdue') dbKey = "OVERDUE_DAYS";
      if (key === 'freeze_days') dbKey = "FREEZE_DAYS";
      if (key.startsWith('chart')) {
        dbKey = key.replace(/([A-Z])/g, '_$1').toUpperCase();
      }
      if (d.useDirectKeys) dbKey = key;

      configMap.set(dbKey, value);
    }

    sheet.clear();
    let rowsToSave = [["Key", "Value"]];
    for (const [k, v] of configMap.entries()) {
      rowsToSave.push([k, sanitizeInput(v)]); // 🔒 XSS protection & batched row payload
    }
    // Write entire config in a single API call (Solves execution limits)
    sheet.getRange(1, 1, rowsToSave.length, 2).setValues(rowsToSave);
    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success" });
  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

function manageInput(token,action, type, value) {
  validateSession(token);
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("Input");
    if (!sheet) {
      sheet = ss.insertSheet("Input");
      sheet.appendRow(["SAC", "UNIT", "EXPENSE_HEAD"]);
    }

    let col = 1;
    if (type === "UNIT") col = 2;
    if (type === "HEAD") col = 3;

    const lastRow = sheet.getLastRow();
    if (action === "add") {
      const data = sheet.getRange(1, col, Math.max(lastRow, 1), 1).getValues();
      for (let i = 0; i < data.length; i++) {
        if (String(data[i][0]).trim().toLowerCase() === String(value).trim().toLowerCase()) {
          return JSON.stringify({ status: "exists" });
        }
      }

      let foundEmpty = false;
      const fullCol = sheet.getRange(1, col, sheet.getMaxRows(), 1).getValues();
      for (let i = 1; i < fullCol.length; i++) {
        if (fullCol[i][0] == "") {
          sheet.getRange(i + 1, col).setValue(value);
          foundEmpty = true;
          break;
        }
      }
      if (!foundEmpty) {
        let lastRowInCol = 0;
        for (let i = fullCol.length - 1; i >= 0; i--) {
          if (fullCol[i][0] != "") { lastRowInCol = i + 1; break; }
        }
        sheet.getRange(lastRowInCol + 1, col).setValue(value);
      }
    } else if (action === "delete") {
      const data = sheet.getRange(1, col, Math.max(lastRow, 1), 1).getValues();
      for (let i = 0; i < data.length; i++) {
        if (String(data[i][0]).trim().toLowerCase() === String(value).trim().toLowerCase()) {
          sheet.getRange(i + 1, col).clearContent();
          break;
        }
      }
    }
    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success" });
  } finally { lock.releaseLock(); }
}

function savePayment(token, formStr) {

    validateSession(token);

    const d = JSON.parse(formStr);

    // 2. AUDIT COMPLIANCE: Dynamic Financial Year Time-Lock
    const todayForFy = new Date();
    const fyStartYear = todayForFy.getMonth() >= 3 ? todayForFy.getFullYear() : todayForFy.getFullYear() - 1;
    const currentFYStart = new Date(fyStartYear, 3, 1);
    const transDate = new Date(d.date);
    if (transDate < currentFYStart) {
      throw new Error(`Financial Year Locked. Cannot create or edit transactions prior to April ${fyStartYear}.`);
    }

    if (typeof hardValidate === "function") hardValidate(d, d.type);
    if (typeof checkPeriodLocks === "function") checkPeriodLocks(d.date);

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Payments");
    if (sheet.getMaxColumns() < 14) {
      sheet.insertColumnsAfter(sheet.getMaxColumns(), 14 - sheet.getMaxColumns());
      sheet.getRange(1, 11).setValue("LastModified");
      sheet.getRange(1, 12).setValue("IsDeleted");
      sheet.getRange(1, 13).setValue("ReceiptURL");
      sheet.getRange(1, 14).setValue("WO_No"); // 🚀 NEW HEADER
    }

   // 🛡️ XSS ARMOR ADDED
    let siteName = sanitizeInput(d.site || "");
    let expenseHead = sanitizeInput(d.head || "");
    let ref = sanitizeInput(d.ref);
    let receiptUrl = sanitizeInput(d.receiptUrl || "");

    // 3. SMART SEQUENTIAL ID FOR PAYMENTS
    let id = d.id;
    if (!id || id === "") {
      const seqPayData = sheet.getDataRange().getValues();
      let maxPay = 0;
      for (let i = 1; i < seqPayData.length; i++) {
        let m = String(seqPayData[i][0]).trim().match(/^PAY-(\d{1,3})$/);
        if (m) maxPay = Math.max(maxPay, parseInt(m[1], 10));
      }
      id = "PAY-" + String(maxPay + 1).padStart(3, '0');
    }

    const newLastModified = new Date().getTime().toString();
    let absAmt = Math.abs(parseFloat(d.amount));

    // 4. PREMIUM AUDIT FIX: Database-Level Overpayment Prevention
    if (d.type === "Invoice" && ref) {
      const invSheet = ss.getSheetByName("Invoices");
      const invData = invSheet.getDataRange().getValues();
      let targetInvTotal = 0;
      let isFound = false;
      for (let i = 1; i < invData.length; i++) {
        if (String(invData[i][6]).trim() === ref && invData[i][38] !== "TRUE") {
          targetInvTotal = parseFloat(invData[i][29]) || 0;
          isFound = true; break;
        }
      }
      if (isFound) {
        const payData = sheet.getDataRange().getValues();
        let alreadyPaid = 0;
        for (let j = 1; j < payData.length; j++) {
          if (payData[j][11] !== "TRUE" && String(payData[j][4]).trim() === ref && String(payData[j][0]) !== String(id)) {
            alreadyPaid += parseFloat(payData[j][5]) || 0;
          }
        }
        const remainingDue = targetInvTotal - alreadyPaid;
        const config = typeof getConfig === "function" ? getConfig() : { PAYMENT_TOLERANCE: 5 };
        const tolerance = parseFloat(config.PAYMENT_TOLERANCE) || 5;

        if (absAmt > (remainingDue + tolerance)) {
          throw new Error(`OVERPAYMENT BLOCKED: Invoice ${ref} only has ₹${remainingDue.toFixed(2)} due. You are trying to apply ₹${absAmt}.`);
        }
      }
    }

    // 5. CONSUME ADVANCE LOGIC
    if (d.type === "Consume Advance") {
      const row = [id, d.date, d.clientName, "Consume Advance", ref, 0, "Journal", siteName, "Advance Applied", `Offset ${absAmt}`, newLastModified, "", ""];
      sheet.appendRow(row);
      if (typeof manageJournalEntry === "function") {
        manageJournalEntry("PAYMENT_ADV_OFFSET", ref, d.date, [
          { acc: d.clientName + " (AR/Adv)", dr: absAmt, cr: 0 },
          { acc: d.clientName + " (AR)", dr: 0, cr: absAmt }
        ]);
      }
      if (typeof logAction === "function") logAction("CONSUME_ADVANCE", absAmt);
      CacheService.getScriptCache().remove("appData_final_chunks");
      return JSON.stringify({ status: "success", message: `₹${absAmt} Advance Consumed for ${ref}` });
    }

    // Format locations/heads
    if (d.type === "GST Payment") {
      siteName = "GST Authority";
      if (d.challan) ref = d.challan;
    } else if (d.type === "Vendor Payment") {
      siteName = d.site || "General Expense";
    } else if (!siteName) {
      const cSheet = ss.getSheetByName("Clients");
      if (cSheet) {
        const cData = cSheet.getDataRange().getValues();
        for (let i = 1; i < cData.length; i++) {
          if (String(cData[i][1]).trim() === String(d.clientName).trim()) {
            siteName = cData[i][8] || "Main Site";
            if (d.site && d.site === cData[i][8]) break;
          }
        }
      }
      if (!siteName) siteName = "Main Site";
    }

    const isNewRecord = (!d.id || d.id === "" || d.id.startsWith("GSTPAY"));
    if (isNewRecord && (d.type === "Advance" || d.type === "Vendor Payment") && (!ref || ref.trim() === "")) {
      const data = sheet.getDataRange().getDisplayValues();
      let maxAdv = 0;
      for (let i = 1; i < data.length; i++) {
        const val = String(data[i][4] || "").toUpperCase().trim();
        if (val.includes("ADV")) {
          const matches = val.match(/(\d+)/);
          if (matches) {
            const num = parseInt(matches[0], 10);
            if (!isNaN(num)) maxAdv = Math.max(maxAdv, num);
          }
        }
      }
      ref = "ADV-" + String(maxAdv + 1).padStart(2, '0');
    }

    // 6. SAVE TO DATABASE
    const pData = sheet.getDataRange().getValues();
    let foundExistingIdx = -1;
    if (d.id && !isNewRecord) {
      for (let i = 1; i < pData.length; i++) {
        if (String(pData[i][0]) === String(d.id)) {
          foundExistingIdx = i; break;
        }
      }
    } else if (d.id && d.id.startsWith("GSTPAY")) {
      for (let i = 1; i < pData.length; i++) {
        if (String(pData[i][0]) === String(d.id)) {
          foundExistingIdx = i; break;
        }
      }
    }

    if (foundExistingIdx > -1) {
      let i = foundExistingIdx;
      let currentLastModified = String(pData[i][10] || "").trim();
      if (d.lastModified && currentLastModified && currentLastModified !== d.lastModified) {
        throw new Error("Concurrency Error: This payment was modified by another user.");
      }

      if (typeof logDetailedAudit === "function") logDetailedAudit("Payments", "EDIT", id, pData[i], d);
      const originalRef = pData[i][4];
      if (!ref || ref.trim() === "" || ref.trim() === "-" || ref === "undefined") {
        ref = originalRef;
      }
      if (!receiptUrl && pData[i][12]) receiptUrl = pData[i][12];
      const row = [id, d.date, d.clientName, d.type, ref, d.amount, d.mode, siteName, expenseHead, "", newLastModified, "", receiptUrl, d.woNo || ""];
      sheet.getRange(i + 1, 1, 1, row.length).setValues([row]);
    } else {
      const row = [id, d.date, d.clientName, d.type, ref, d.amount, d.mode, siteName, expenseHead, "", newLastModified, "", receiptUrl, d.woNo || ""];
      sheet.appendRow(row);
      if (typeof logDetailedAudit === "function") logDetailedAudit("Payments", "CREATE", id, {}, d);
    }

    // Journal Entries
    let drAcc, crAcc;
    if (d.type === "GST Payment") {
      drAcc = "GST Payable (Offset)";
      crAcc = "Bank/Cash (" + d.mode + ")";
    } else if (d.type === "Vendor Payment" || (d.type === "Expense" && parseFloat(d.amount) < 0)) {
      drAcc = d.clientName + " (AP/Expense)";
      crAcc = "Bank/Cash (" + d.mode + ")";
    } else if (parseFloat(d.amount) > 0) {
      drAcc = "Bank/Cash (" + d.mode + ")";
      crAcc = d.clientName + " (AR/Adv)";
    } else {
      drAcc = d.clientName + " (AP)";
      crAcc = "Bank/Cash (" + d.mode + ")";
    }
    if (typeof manageJournalEntry === "function") {
      manageJournalEntry("PAYMENT", ref || id, d.date, [
        { acc: drAcc, dr: absAmt, cr: 0 },
        { acc: crAcc, dr: 0, cr: absAmt }
      ]);
    }

    // =======================================================
    // 🚀 7. NEW: INTEGRATED LOGGING & CACHE FLUSH 
    // =======================================================
    if (typeof logAction === "function") logAction("ADD_EDIT_PAYMENT", absAmt);

    SpreadsheetApp.flush(); // Forces Google to write the data immediately
    CacheService.getScriptCache().remove("appData_final_chunks"); // Clears the UI cache

    return JSON.stringify({ status: "success", message: "Saved. Ref: " + ref });

  } catch (e) {
    if (typeof logSystemError === "function") logSystemError("savePayment", e.message);
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

function getVendorLedger(token, vendorName, startDate, endDate, siteFilter) {
  validateSession(token);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const payments = ss.getSheetByName("Payments").getDataRange().getDisplayValues();
  const toYMD = (dStr) => {
    if (!dStr) return 0;
    if (dStr instanceof Date) {
      return parseInt(Utilities.formatDate(dStr, Session.getScriptTimeZone(), "yyyyMMdd"));
    }
    const parts = String(dStr).split('-');
    if (parts.length === 3) {
      return parseInt(parts[0] + parts[1] + parts[2]);
    }
    return 0;
  };

  const startVal = toYMD(startDate || '2000-01-01');
  const endVal = toYMD(endDate || '2099-12-31');
  const targetVendor = String(vendorName).trim().toLowerCase();
  const targetSite = siteFilter && siteFilter !== "" ? String(siteFilter).trim().toLowerCase() : null;

  let txns = [];
  for (let i = 1; i < payments.length; i++) {
    const r = payments[i];
    if (r[11] === "TRUE") continue;
    const rowDateVal = toYMD(r[1]);
    const rowName = String(r[2]).trim().toLowerCase();
    const rowSite = String(r[7] || "").trim().toLowerCase();
    if (rowName === targetVendor && rowDateVal >= startVal && rowDateVal <= endVal) {
      if (targetSite && rowSite !== targetSite) continue;
      const head = r[8] ? r[8] + " - " : "";
      const raw = String(r[5]).replace(/[^0-9.-]+/g, "");
      const amt = Math.abs(parseFloat(raw) || 0);
      let refStr = r[4] ? `(Ref: ${r[4]})` : `(Ref: -)`;
      let siteStr = r[7] ? `[Site: ${r[7]}]` : `[Site: Main Site]`;
      let descStr = `Payment: ${head}${refStr} ${siteStr}`;
      txns.push({
        date: r[1],
        desc: descStr,
        debit: amt,
        credit: 0,
        site: r[7]
      });
    }
  }
  return JSON.stringify(txns.sort((a, b) => toYMD(a.date) - toYMD(b.date)));
}

function getLedgerData(token, clientName, clientSite, sDate, eDate) {
  validateSession(token);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const invData = ss.getSheetByName("Invoices").getDataRange().getDisplayValues();
  const payData = ss.getSheetByName("Payments").getDataRange().getDisplayValues();
  const jSheet = ss.getSheetByName("Journal"); // Added to read manual journals

  let txns = [];
  const start = sDate ? new Date(sDate) : new Date('2000-01-01');
  const end = eDate ? new Date(eDate) : new Date('2099-12-31');
  const hasSite = (clientSite && clientSite !== "" && clientSite !== "All Sites");
  // 1. Get Invoices
  invData.slice(1).forEach(r => {
    if (r[38] === "TRUE") return;
    const isCancelled = (r[34] && r[34].includes("CANCELLED_INV")) || (r[36] === "CANCELLED");

    // FIX: ONLY allow Tax Invoices into the Ledger!
    if (r[5] === clientName && r[2] === "Tax Invoice" && !isCancelled) {
      if (hasSite && r[4] !== clientSite) return;
      const d = new Date(r[7]);
      if (d >= start && d <= end) txns.push({ date: r[7], desc: "Inv: " + r[6], debit: parseFloat(r[29] || 0), credit: 0 });
    }
  });

  // 2. Get Payments
  payData.slice(1).forEach(r => {
    if (r[11] === "TRUE") return;
    if (r[2] === clientName) {
      if (hasSite && r[7] !== clientSite) return;
      const d = new Date(r[1]);
      if (d >= start && d <= end) {
        let modeStr = r[6] ? " (" + r[6] + ")" : "";
        let headStr = r[8] ? " [" + r[8] + "]" : "";

        // 🚀 NEW: Pull the tracking history from the Remarks (Column J / r[9])
        let auditStr = r[9] ? " {" + r[9] + "}" : "";

        txns.push({
          date: r[1],
          desc: "Payment: " + r[4] + modeStr + headStr + auditStr,
          debit: 0,
          credit: parseFloat(r[5] || 0)
        });
      }
    }
  });

  // 3. Get Manual Journals (Fixes the Ledger Blindspot)
  if (jSheet) {
    const jData = jSheet.getDataRange().getDisplayValues();
    const targetAR = clientName + " (AR)";
    const targetAP = clientName + " (AP)";
    const targetAdv = clientName + " (AR/Adv)";

    jData.slice(1).forEach(r => {
      if (r[8] === "TRUE" || r[8] === true) return; // Skip reversed/deleted entries
      if (r[1] !== "MANUAL_JOURNAL") return; // Only pull Manual Journals (Invoices/Payments handled above)

      let acc = String(r[5]).trim();
      if (acc === targetAR || acc === targetAP || acc === targetAdv) {
        const d = new Date(r[3]);
        if (d >= start && d <= end) {
          txns.push({
            date: r[3],
            desc: "Manual Journal: " + r[2], // Shows the Manual Journal ID
            debit: parseFloat(r[6]) || 0,
            credit: parseFloat(r[7]) || 0
          });
        }
      }
    });
  }

  return JSON.stringify(txns.sort((a, b) => new Date(a.date) - new Date(b.date)));
}

function cancelInvoice(token, id) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    validateSession(token);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Invoices");
    const data = sheet.getDataRange().getDisplayValues();

    if (sheet.getMaxColumns() < 39) {
      sheet.insertColumnsAfter(sheet.getMaxColumns(), 39 - sheet.getMaxColumns());
      sheet.getRange(1, 37).setValue("Cancel_Status");
      sheet.getRange(1, 38).setValue("LastModified");
      sheet.getRange(1, 39).setValue("IsDeleted");
    }

    for (let i = 1; i < data.length; i++) {
      // --- CROSS-SHEET PAYMENT GUARDRAIL ---
      // The 'id' here is the Invoice Number (e.g., SD/25-26/001)
      const paySheet = ss.getSheetByName("Payments");
      if (paySheet) {
        const payData = paySheet.getDataRange().getValues();
        let totalPaid = 0;

        for (let p = 1; p < payData.length; p++) {
          // Check if the payment itself isn't deleted (Index 11 is IsDeleted)
          let isPayDeleted = (String(payData[p][11]).trim().toUpperCase() === "TRUE");
          let payRef = String(payData[p][4]).trim(); // Index 4 is 'Ref' (Invoice No)
          let payAmount = parseFloat(payData[p][5]) || 0; // Index 5 is 'Amount'

          if (!isPayDeleted && payRef === String(id).trim()) {
            totalPaid += payAmount;
          }
        }

        if (totalPaid > 0) {
          return JSON.stringify({
            status: "error",
            message: `Security Block: Cannot cancel. ₹${totalPaid} has already been received for Invoice ${id}. Please delete the payment receipt first.`
          });
        }
      }
      // -------------------------------------
      if (data[i][6] === id) {
        const invDate = data[i][7];
        checkPeriodLocks(invDate);

        const currentStatus = String(data[i][36] || "").trim().toUpperCase();

        // --- NEW SAFETY GUARDRAIL: Prevent Cancelling PAID Invoices ---
        if (currentStatus === "PAID" || currentStatus === "RECEIVED") {
          return JSON.stringify({ status: "error", message: "Security Block: Cannot cancel an invoice that has already been PAID. Please reverse/delete the associated payment receipt first." });
        }
        // --------------------------------------------------------------

        if (currentStatus !== "CANCELLED") {
          sheet.getRange(i + 1, 37).setValue("CANCELLED");
          try { manageJournalEntry("INVOICE", id, invDate, []); } catch (e) { }

          logAction("CANCEL_INVOICE", id);
          logDetailedAudit("Invoices", "CANCEL", id, data[i], { status: "CANCELLED" });
          CacheService.getScriptCache().remove("appData_final_chunks");
          return JSON.stringify({ status: "success" });
        }
      }
    }
    return JSON.stringify({ status: "error", message: "Invoice not found" });
  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

  } finally {

function cleanAuditLog(token,days) {
  validateSession(token);
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("AuditLog");
    if (!sheet) return "No AuditLog found";
    const safeDays = parseInt(days) || 30;
    const cutoff = new Date();
    cutoff.setDate(cutoff.getDate() - safeDays);
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return "Log already empty or only header.";

    const rowsToKeep = [data[0]];
    let deletedCount = 0;
    for (let i = 1; i < data.length; i++) {
      const rowDate = new Date(data[i][0]);
      if (rowDate >= cutoff) {
        rowsToKeep.push(data[i]);
      } else {
        deletedCount++;
      }
    }

    if (deletedCount > 0) {
      sheet.clearContents();
      sheet.getRange(1, 1, rowsToKeep.length, rowsToKeep[0].length).setValues(rowsToKeep);
    }
    return `Cleaned ${deletedCount} logs older than ${safeDays} days.`;
  } catch (e) { return e.toString(); } finally { lock.releaseLock(); }
}

// SAVE INVOICE (With Smart Work Order Linker)(Premium UPI Generation & Multiple Templates)

function saveInvoice(token, formStr) { // 👈 1. Added token parameter
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);

    // ========================================================
    // 🛑 1. THE GATEKEEPER: SECURITY & TIME-LOCKS
    // These MUST happen before touching the database!
    // ========================================================
    if (!token || token.length > 250) throw new Error("Security Block: Invalid or missing token.");
    validateSession(token); // 🔒 STRICT AUTH ENFORCEMENT

    const d = JSON.parse(formStr);

    // A. Financial Year Lock (No Time-Travel)
    const todayForFy = new Date();
    const fyStartYear = todayForFy.getMonth() >= 3 ? todayForFy.getFullYear() : todayForFy.getFullYear() - 1;
    const currentFYStart = new Date(fyStartYear, 3, 1);
    const transDate = new Date(d.date);

    if (transDate < currentFYStart) {
      return JSON.stringify({
        status: "error",
        message: `Financial Year Locked. Cannot create or edit transactions prior to April 1, ${fyStartYear}.`
      });
    }

    // B. Core Validations & Period Locks
    if (typeof hardValidate === "function") hardValidate(d, d.invType);
    if (typeof checkPeriodLocks === "function") checkPeriodLocks(d.date);

    const valResult = validateGSTINBackend(d.gstin, d.clientState);
    if (!valResult.valid) return JSON.stringify({ status: "error", message: valResult.message });

    // ========================================================
    // 🟢 2. DATABASE READ/WRITES (Safe to proceed)
    // ========================================================

    // --- STRICT WORK ORDER LIMIT VALIDATION ---
    if (d.soNumber && d.soNumber.trim() !== "") {
      const woSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("WorkOrders");
      if (woSheet) {
        const woData = woSheet.getDataRange().getValues();
        let woTotalAmt = 0;
        let woFound = false;

        // 1. Find the Work Order Total
        for (let w = 1; w < woData.length; w++) {
          if (String(woData[w][0]).trim().toUpperCase() === String(d.soNumber).trim().toUpperCase()) {
            woFound = true;
            try {
              let woItems = JSON.parse(woData[w][9] || "[]");
              woItems.forEach(it => { woTotalAmt += parseFloat(it.amount) || 0; });
            } catch (e) { }
            break;
          }
        }

        // 2. If WO exists, calculate how much has ALREADY been billed against it
        if (woFound && woTotalAmt > 0) {
          const invSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Invoices");
          const invData = invSheet.getDataRange().getValues();
          let pastBilledAmt = 0;

          for (let j = 1; j < invData.length; j++) {
            if (invData[j][38] === "TRUE") continue; // Skip deleted
            
            // 🚀 NEW FIX: ONLY count actual Tax Invoices against the Work Order limit!
            let docType = String(invData[j][2]).trim();
            if (docType === "Proforma Invoice" || docType === "Quotation" || docType === "Delivery Challan") continue;

            let existingStatus = String(invData[j][36]).trim() || String(invData[j][34]).trim();
            if (existingStatus.includes("CANCELLED")) continue; // Skip cancelled

            // Skip the current invoice if we are editing an existing one
            if (String(invData[j][6]).trim() === String(d.invNum).trim()) continue;

            if (String(invData[j][10]).trim().toUpperCase() === String(d.soNumber).trim().toUpperCase()) {
              try {
                let invItems = JSON.parse(invData[j][32] || "[]");
                invItems.forEach(it => { pastBilledAmt += parseFloat(it.amount) || 0; });
              } catch (e) { }
            }
          }

          // 3. Calculate this exact invoice's total
          let currentBilledAmt = 0;
          try {
            let currItems = JSON.parse(d.itemsJson || "[]");
            currItems.forEach(it => { currentBilledAmt += parseFloat(it.amount) || 0; });
          } catch (e) { }

          // 4. BLOCK if it exceeds (with a ₹5 tolerance for rounding decimals)
          if ((pastBilledAmt + currentBilledAmt) > (woTotalAmt + 5)) {
            return JSON.stringify({
              status: "error",
              message: `WORK ORDER LIMIT EXCEEDED!\n\nWO Total: ₹${woTotalAmt.toFixed(2)}\nPreviously Billed: ₹${pastBilledAmt.toFixed(2)}\nThis Invoice: ₹${currentBilledAmt.toFixed(2)}\n\nYou cannot exceed the approved Work Order amount.`
            });
          }
        }
      }
    }
    // --- END WO VALIDATION ---

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Invoices");
    if (sheet.getMaxColumns() < 39) {
      sheet.insertColumnsAfter(sheet.getMaxColumns(), 39 - sheet.getMaxColumns());
      sheet.getRange(1, 37).setValue("Cancel_Status");
      sheet.getRange(1, 38).setValue("LastModified");
      sheet.getRange(1, 39).setValue("IsDeleted");
    }

    const data = sheet.getDataRange().getDisplayValues();
    let foundRow = -1;
    let oldPdfUrl = "";
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][6]).trim() === String(d.invNum).trim() && data[i][38] !== "TRUE") {
        foundRow = i + 1;
        oldPdfUrl = data[i][30];
        break;
      }
    }

    let isDuplicate = false;
    for (let j = 1; j < data.length; j++) {
      if (data[j][38] === "TRUE") continue;
      let existingInvNum = String(data[j][6]).trim();
      let existingStatus = String(data[j][36]).trim() || String(data[j][34]).trim();
      if (existingInvNum.toUpperCase() === String(d.invNum).trim().toUpperCase() && j !== (foundRow - 1) && !existingStatus.includes("CANCELLED")) {
        isDuplicate = true;
        break;
      }
    }
    if (isDuplicate && foundRow === -1) {
      return JSON.stringify({ status: "error", message: "Duplicate Invoice Blocked! Please use a unique invoice number." });
    }

    let currentLastModified = "";
    if (foundRow > -1) {
      // STRICT BLOCK: If the invoice exists but the user's form thinks it's brand new, BLOCK IT.
      if (!d.lastModified || d.lastModified.trim() === "") {
        return JSON.stringify({ status: "error", message: "Duplicate Invoice Blocked! Another user has already generated this invoice number. Please refresh." });
      }

      const originalInvNum = String(data[foundRow - 1][6]).trim();
      if (originalInvNum !== String(d.invNum).trim()) {
        return JSON.stringify({ status: "error", message: "Enterprise Security: Modifying the Invoice Number of an existing record breaks payment links. Please cancel this invoice and create a new one." });
      }
      currentLastModified = String(data[foundRow - 1][37] || "").trim();
      if (d.lastModified && currentLastModified && currentLastModified !== d.lastModified) {
        return JSON.stringify({ status: "error", message: "Concurrency Error: This invoice was modified by another user since you opened it. Please refresh and try again." });
      }
      try {
        let revSheet = ss.getSheetByName("InvoiceRevisions");
        if (!revSheet) { revSheet = ss.insertSheet("InvoiceRevisions"); }
        revSheet.appendRow([new Date().toISOString(), Session.getActiveUser().getEmail(), "VER_BKP", ...data[foundRow - 1]]);
      } catch (e) { }

      if (typeof logDetailedAudit === "function") logDetailedAudit("Invoices", "EDIT", d.invNum, data[foundRow - 1], d);
    } else {
      if (typeof logDetailedAudit === "function") logDetailedAudit("Invoices", "CREATE", d.invNum, {}, d);
    }
    const newLastModified = new Date().getTime().toString();

    const config = getConfig();
    const folderName = config.DRIVE_FOLDER || DEFAULT_DRIVE_FOLDER;
    let signatureUrl = config.SIGNATURE_URL || "";
    if (signatureUrl.includes("drive.google.com") && signatureUrl.includes("/d/")) {
      try {
        const id = signatureUrl.split('/d/')[1].split('/')[0];
        const imgBlob = DriveApp.getFileById(id).getBlob();

        const mime = imgBlob.getContentType();
        signatureUrl = `data:${mime};base64,${base64}`;

        try {
          const id = signatureUrl.split('/d/')[1].split('/')[0];
          signatureUrl = "https://drive.google.com/uc?export=view&id=" + id;
        } catch (ex) { }
      }

    d.upiId = config.UPI_ID || "";

    const folders = DriveApp.getFoldersByName(folderName);
    if (folders.hasNext()) folder = folders.next();
    else folder = DriveApp.createFolder(folderName);

        const oldFileId = oldPdfUrl.match(/id=([a-zA-Z0-9_-]+)/)[1];
        DriveApp.getFileById(oldFileId).setTrashed(true);
      } catch (e) { }
    }

    const existingFiles = folder.getFilesByName(fileName);
    while (existingFiles.hasNext()) {
      try { existingFiles.next().setTrashed(true); } catch (e) { }
    }

    const secretSalt = "SAIDURGA-SECURE-2026";
    const rawData = `${d.invNum}|${d.date}|${d.clientName}|${d.grandTotal}|${secretSalt}`;
    const hashBytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, rawData);
    d.digitalFingerprint = hashBytes.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');

    const html = createInvoiceHtmlRouter(d);
    const blob = Utilities.newBlob(html, 'text/html', "INV-" + d.invNum + ".html");
    let pdf;
    try {
      pdf = folder.createFile(blob.getAs(MimeType.PDF)).setName(fileName);
    } catch (e) {
      pdf = { getUrl: () => "#" };
    }

    const row = [
      "UID-" + Date.now(), "Tax Invoice", d.invType, d.clientState, d.siteName, d.clientName, d.invNum, d.date,
      d.billFrom, d.billTo, d.soNumber, d.billingAddress, d.gstin, d.gstRate, d.vehicle,

      "1", "", "", "", "", "", "", d.igst, d.cgst, d.sgst, d.subTotal, d.grandTotal, d.words, pdf.getUrl(),
      d.itemsJson, d.roundOff, d.terms, "", "", newLastModified, ""
    ];

    if (foundRow > -1) {
      row[36] = data[foundRow - 1][36] || "";
      sheet.getRange(foundRow, 1, 1, row.length).setValues([row]);
    } else {
      sheet.appendRow(row);
    }

    try {

      if (d.invType === "Tax Invoice") {
        manageJournalEntry("INVOICE", d.invNum, d.date, [

          { acc: "IGST Payable", dr: parseFloat(d.igst), cr: 0 },
          { acc: "CGST Payable", dr: parseFloat(d.cgst), cr: 0 },
          { acc: "SGST Payable", dr: parseFloat(d.sgst), cr: 0 },
          { acc: "Round Off", dr: roundOffVal > 0 ? roundOffVal : 0, cr: roundOffVal < 0 ? Math.abs(roundOffVal) : 0 },
          { acc: d.clientName + " (AR)", dr: 0, cr: parseFloat(d.grandTotal) }
        ]);
      }
    } catch (journalErr) { console.error("Journal Post Failed"); }

    // SMART WORK ORDER LINKING (PARTIAL vs COMPLETED)
    // ==========================================
    if (d.soNumber && d.soNumber.trim() !== "") {

        const woSheet = ss.getSheetByName("WorkOrders");
        if (woSheet) {
          const woData = woSheet.getDataRange().getValues();
          let woFound = false;

          let totalBilledAmt = 0;
          for (let j = 1; j < data.length; j++) {
            if (data[j][38] === "TRUE") continue;
            let existingStatus = String(data[j][36]).trim() || String(data[j][34]).trim();
            if (existingStatus.includes("CANCELLED")) continue;

            if (String(data[j][10]).trim().toUpperCase() === String(d.soNumber).trim().toUpperCase()) {

                if (totalBilledAmt >= (woTotalAmt - 1)) {
                  newStatus = "Completed";
                } else if (totalBilledAmt > 0) {
                  newStatus = "Partial";
                } else {
                  newStatus = (currentStatus === "Completed" || currentStatus === "Partial") ? "In Progress" : currentStatus;
                }
              } else {
                newStatus = "Completed";
              }

              woSheet.getRange(w + 1, 8).setValue(newStatus);
              break;
            }
          }

          if (!woFound) {
            const todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
            woSheet.appendRow([d.soNumber, d.clientName, d.siteName, d.workDesc, "Auto-Linked", todayStr, todayStr, "Completed", pdf.getUrl(), "[]"]);
          }
        }
      } catch (woErr) { console.error("WO Sync Error:", woErr); }
    }
    // --- END WO VALIDATION ---

    // ========================================================
    // 🚀 PREMIUM AUTO-LINK ENGINE (Fixed for Exact Math)
    // ========================================================
    if (d.convertedFrom && String(d.convertedFrom).trim() !== "") {
      try {
        const oldDocNum = String(d.convertedFrom).trim();
        const newTaxNum = String(d.invNum).trim();

        if (paySheet) {

            if (payRef === oldDocNum && !isDeleted) {
              // 1. STRICT MATCH FOR MATH: Keep the reference exactly the new invoice number
              paySheet.getRange(i + 1, 5).setValue(newTaxNum);

              // 2. AUDIT TRAIL: Put the history strictly in the Remarks column (Column J)
              let existingRemarks = String(payData[i][9] || "").trim();
              let newRemarks = existingRemarks ? existingRemarks + " | Auto-Linked from: " + oldDocNum : "Auto-Linked from: " + oldDocNum;
              paySheet.getRange(i + 1, 10).setValue(newRemarks);
            }
          }
        }

        // 3. Safely Lock the Old Document
        const invData = sheet.getDataRange().getValues();
        for (let k = 1; k < invData.length; k++) {
          if (String(invData[k][6]).trim() === oldDocNum && invData[k][38] !== "TRUE") {
            sheet.getRange(k + 1, 37).setValue("CONVERTED TO " + newTaxNum);
            break;
          }
        }

      } catch (err) {
        console.error("Auto-Link & Lock Failed: ", err.message);
      }
    }
    // ========================================================

    CacheService.getScriptCache().remove("appData_final_chunks");

    return JSON.stringify({ status: "success", url: pdf.getUrl() });
  } catch (e) {
    if (typeof logSystemError === "function") logSystemError("saveInvoice", e.message);
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

function saveClient(token, formStr) {  // 👈 Added token parameter
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    validateSession(token); // 🔒 STRICT AUTH ENFORCEMENT

    const d = JSON.parse(formStr);

    // 🛡️ XSS ARMOR: Strip HTML tags before saving
    if (d.name) d.name = sanitizeInput(d.name);
    if (d.address) d.address = sanitizeInput(d.address);
    if (d.city) d.city = sanitizeInput(d.city);
    // Loosened GSTIN Regex to prevent false checksum errors, but enforces 15 chars
    if (d.gstin && d.gstin.length !== 15 && d.gstin.toUpperCase() !== "N/A") {
      return JSON.stringify({ status: "error", message: "Invalid Format: GSTIN must be exactly 15 characters long." });
    }
    const val = validateGSTINBackend(d.gstin, d.state);
    if (!val.valid && d.gstin.length === 15) return JSON.stringify({ status: "error", message: val.message });

    const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Clients");
    const dt = sheet.getDataRange().getDisplayValues();

    // SECURITY: Block duplicate Client Name + Site combo
    if (!d.id) {
      for (let i = 1; i < dt.length; i++) {
        if (dt[i][1].trim().toLowerCase() === d.name.trim().toLowerCase() && (dt[i][8] || "Main Site").trim().toLowerCase() === (d.site || "Main Site").trim().toLowerCase()) {
          return JSON.stringify({ status: "error", message: "Duplicate Blocked: A client with this exact Name and Site Name already exists!" });
        }
      }
    }

    let newId = d.id; if (!newId) {
      const siteKey = (d.site || d.name).substring(0, 3).toUpperCase();
      const data = sheet.getRange(2, 1, Math.max(sheet.getLastRow() - 1, 1), 1).getDisplayValues(); let maxNum = 0;
      data.forEach(r => { const existingId = String(r[0]); if (existingId.startsWith(siteKey + "-")) { const numPart = parseInt(existingId.split("-")[1]); if (!isNaN(numPart)) maxNum = Math.max(maxNum, numPart); } }); newId = `${siteKey}-${String(maxNum + 1).padStart(2, '0')}`;
    } const row = [newId, d.name, d.gstin, d.state, d.city, d.address, d.email, d.phone, d.site];
    if (d.id) {
      for (let i = 1; i < dt.length; i++) {
        if (dt[i][0] == d.id) {
          sheet.getRange(i + 1, 1, 1, row.length).setValues([row]);
          CacheService.getScriptCache().remove("appData_final_chunks"); return JSON.stringify({ status: "success" });
        }
      }
    }
    sheet.appendRow(row);
    CacheService.getScriptCache().remove("appData_final_chunks"); return JSON.stringify({ status: "success" });
  } finally { lock.releaseLock(); }
}

function getPendingInvoices(clientName, clientSite) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const invData = ss.getSheetByName("Invoices").getDataRange().getDisplayValues();
  const payData = ss.getSheetByName("Payments").getDataRange().getDisplayValues();
  let payMap = {};
  payData.forEach(r => {
    if (r[11] === "TRUE") return;
    if (r[4]) payMap[r[4]] = (payMap[r[4]] || 0) + parseFloat(r[5] || 0);
  });
  let pending = [];
  const targetName = String(clientName).trim().toLowerCase();
  invData.slice(1).forEach(r => {
    if (r[38] === "TRUE") return;
    const rowName = String(r[5] || "").trim().toLowerCase();
    if (rowName === targetName && r[6] && r[2] !== "Quotation" && r[2] !== "Delivery Challan") {
      const d = parseFloat(r[29] || 0) - (payMap[r[6]] || 0);

  });

function getGSTReportData(token,month, year) {
  validateSession(token);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const data = ss.getSheetByName("Invoices").getDataRange().getDisplayValues();
  const b2b = [], b2c = [], sacMap = {};

  for (let i = 1; i < data.length; i++) {
    const r = data[i];

    // 1. Skip empty rows or deleted rows
    if (!r[7] || r[38] === "TRUE") continue;

    // 2. Skip Cancelled Invoices
    let isCancelled = (String(r[34]).includes("CANCELLED") || String(r[36]).includes("CANCELLED"));
    if (isCancelled) continue;

    const rowDate = r[7];
    const parts = rowDate.split('-');
    if (parts.length < 2) continue;

    const rowY = parseInt(parts[0]);
    const rowM = parseInt(parts[1]);

    if (rowM === parseInt(month) && rowY === parseInt(year)) {

      // 🛑 THE FIX: Strictly block non-financial documents (Checking both r[1] and r[2] to be safe)
      const doc1 = String(r[1]).trim();
      const doc2 = String(r[2]).trim();
      if (doc1 === "Proforma Invoice" || doc2 === "Proforma Invoice" ||
        doc1 === "Quotation" || doc2 === "Quotation" ||
        doc1 === "Delivery Challan" || doc2 === "Delivery Challan") {
        continue;
      }
      // Only proceed if it is a "Tax Invoice" (or Credit Note)

      const gstin = r[12] || "";
      const taxable = parseFloat(r[28] || 0);
      const total = parseFloat(r[29] || 0);

      const obj = {
        client: r[5],
        gstin: gstin,
        invNum: r[6],
        date: r[7],
        state: r[3],
        taxable: taxable,
        igst: parseFloat(r[25] || 0),
        cgst: parseFloat(r[26] || 0),
        sgst: parseFloat(r[27] || 0),
        total: total
      };

      if (gstin.length > 2) {
        b2b.push(obj);
      } else {
        b2c.push(obj);
      }

      try {
        if (r[32] && r[32] !== "") {
          JSON.parse(r[32]).forEach(item => {
            if (!sacMap[item.sac]) sacMap[item.sac] = { sac: item.sac, taxable: 0, tax: 0, total: 0 };
            const v = parseFloat(item.amount);
            const rate = parseFloat(r[13] || 18) / 100;
            sacMap[item.sac].taxable += v;
            sacMap[item.sac].tax += (v * rate);
            sacMap[item.sac].total += (v * (1 + rate));
          });
        }
      } catch (e) {
        console.error("SAC Parse Error: ", e);
      }
    }
  }

  return JSON.stringify({ b2b, b2c, sac: Object.values(sacMap) });
}

function recalculateWorkOrderStatus(woNo) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const woSheet = ss.getSheetByName("WorkOrders");
    const invSheet = ss.getSheetByName("Invoices");
    if (!woSheet || !invSheet) return;

    const woData = woSheet.getDataRange().getValues();
    const invData = invSheet.getDataRange().getValues();

    let woRow = -1; let woLimit = 0;
    for (let i = 1; i < woData.length; i++) {
      if (String(woData[i][0]).trim().toUpperCase() === String(woNo).toUpperCase()) {
        woRow = i + 1;
        try {
          let items = JSON.parse(woData[i][9] || "[]");
          items.forEach(it => { woLimit += parseFloat(it.amount) || 0; });
        } catch (e) { }
        break;
      }
    }
    if (woRow === -1) return;

    let activeBilled = 0;
    for (let j = 1; j < invData.length; j++) {
      if (invData[j][38] === "TRUE" || invData[j][38] === true) continue; // Skip deleted
      let status = String(invData[j][36]).trim() || String(invData[j][34]).trim();
      if (status.includes("CANCELLED")) continue; // Skip cancelled

      if (String(invData[j][10]).trim().toUpperCase() === String(woNo).toUpperCase()) {
        try {
          let invItems = JSON.parse(invData[j][32] || "[]");
          invItems.forEach(it => { activeBilled += parseFloat(it.amount) || 0; });
        } catch (e) { }
      }
    }

    let newStatus = woData[woRow - 1][7];
    if (woLimit > 0) {
      if (activeBilled >= (woLimit - 1)) newStatus = "Completed";
      else if (activeBilled > 0) newStatus = "Partial";
      else newStatus = "In Progress";
    }
    woSheet.getRange(woRow, 8).setValue(newStatus);
  } catch (e) { }
}

function getAdvancedReports(token,type, params) {
  validateSession(token);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const rawInvoices = ss.getSheetByName("Invoices").getDataRange().getDisplayValues();
  const rawPayments = ss.getSheetByName("Payments").getDataRange().getDisplayValues();
  let gstPaidMap = {};
  rawPayments.forEach(r => {
    if (r[11] === "TRUE") return;
    if (r[3] === "GST Payment" && r[1]) {
      const d = new Date(r[1]);
      if (!isNaN(d.getTime())) {
        const mKey = d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, '0');
        gstPaidMap[mKey] = (gstPaidMap[mKey] || 0) + Math.abs(parseFloat(r[5]) || 0);
      }
    }
  });
  const invoices = rawInvoices.slice(1).filter(r => r[38] !== "TRUE");
  if (type === "GSTR3B") {
    const m = parseInt(params.month);
    const y = parseInt(params.year);
    let summary = { taxable: 0, igst: 0, cgst: 0, sgst: 0, total: 0, itc: 0 };
    invoices.forEach(r => {
      const d = new Date(r[7]); const invType = r[2];
      const isCancelled = (r[34] && r[34].includes("CANCELLED")) || (r[36] === "CANCELLED");
      if (!isNaN(d.getTime()) && (d.getMonth() + 1) === m && d.getFullYear() === y && invType === "Tax Invoice" && !isCancelled) {
        summary.taxable += parseFloat(r[28]) || 0; summary.igst += parseFloat(r[25]) || 0;
        summary.cgst += parseFloat(r[26]) || 0; summary.sgst += parseFloat(r[27]) || 0; summary.total += parseFloat(r[29]) || 0;
      }
    });
    const rawPurchases = ss.getSheetByName("Purchases");
    if (rawPurchases) {
      const pData = rawPurchases.getDataRange().getDisplayValues();
      for (let i = 1; i < pData.length; i++) {
        const r = pData[i];
        if (r[11] === "TRUE") continue;
        const d = new Date(r[1]);
        if (!isNaN(d.getTime()) && (d.getMonth() + 1) === m && d.getFullYear() === y) {
          const taxAmt = parseFloat(r[6]) || 0;
          summary.itc += taxAmt;
        }
      }
    }
    return JSON.stringify(summary);
  }

  if (type === "HSN") {
    const m = parseInt(params.month);
    const y = parseInt(params.year);
    let hsnMap = {};
    invoices.forEach(r => {
      const d = new Date(r[7]); const invType = r[2];
      const isCancelled = (r[34] && r[34].includes("CANCELLED")) || (r[36] === "CANCELLED");
      if ((!params.month || ((d.getMonth() + 1) === m && d.getFullYear() === y)) && invType === "Tax Invoice" && !isCancelled) {
        try {
          const items = JSON.parse(r[32]);
          items.forEach(item => {
            const code = item.sac || "NA";
            if (!hsnMap[code]) hsnMap[code] = { code: code, count: 0, taxable: 0, gst: 0 };
            const amt = parseFloat(item.amount) || 0;
            hsnMap[code].count++; hsnMap[code].taxable += amt;
          });
        } catch (e) { }
      }
    });
    return JSON.stringify(Object.values(hsnMap));
  }

  if (type === "Aging") {
    let payMap = {};
    rawPayments.forEach(r => {
      if (r[11] === "TRUE") return;
      if (r[4]) payMap[r[4]] = (payMap[r[4]] || 0) + (parseFloat(r[5]) || 0);
    });
    let clientMap = {};
    const today = new Date();

    let config = getConfig();
    let overdueDaysConfig = parseInt(config.OVERDUE_DAYS) || 45;
    invoices.forEach(r => {
      const client = r[5]; const total = parseFloat(r[29]) || 0; const paid = payMap[r[6]] || 0; const due = total - paid;
      const invType = r[2];
      const isCancelled = (r[34] && r[34].includes("CANCELLED")) || (r[36] === "CANCELLED");

      if (invType === "Tax Invoice" && !isCancelled && due > 1) {
        if (!clientMap[client]) clientMap[client] = { name: client, "0-30": 0, "31-60": 0, "61-90": 0, "90+": 0, total: 0 };

        const d = new Date(r[7]);
        const dueDate = new Date(d.getTime() + (overdueDaysConfig * 86400000));
        const age = Math.floor((today - dueDate) / (1000 * 60 * 60 * 24));

        if (age > 0) {
          if (age <= 30) clientMap[client]["0-30"] += due;
          else if (age <= 60) clientMap[client]["31-60"] += due;
          else if (age <= 90) clientMap[client]["61-90"] += due;
          else clientMap[client]["90+"] += due;
          clientMap[client].total += due;
        }
      }
    });
    return JSON.stringify(Object.values(clientMap));
  }

  if (type === "Liability") {
    let liabilityMap = {};
    invoices.forEach(r => {
      const invType = r[2];
      const isCancelled = (r[34] && r[34].includes("CANCELLED")) || (r[36] === "CANCELLED");
      if (invType === "Tax Invoice" && !isCancelled) {
        const d = new Date(r[7]);
        if (!isNaN(d.getTime())) {
          const key = d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, '0');
          const tax = (parseFloat(r[25]) || 0) + (parseFloat(r[26]) || 0) + (parseFloat(r[27]) || 0);

          if (!liabilityMap[key]) liabilityMap[key] = { month: key, liability: 0, itc: 0, net: 0, paid: 0 };
          liabilityMap[key].liability += tax;
        }
      }
    });
    const rawPurchases = ss.getSheetByName("Purchases");
    if (rawPurchases) {
      rawPurchases.getDataRange().getDisplayValues().slice(1).forEach(r => {
        if (r[11] === "TRUE") return;
        const d = new Date(r[1]);
        if (!isNaN(d.getTime())) {
          const key = d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, '0');
          if (!liabilityMap[key]) liabilityMap[key] = { month: key, liability: 0, itc: 0, net: 0, paid: 0 };

          const taxAmt = parseFloat(r[6]) || 0;
          liabilityMap[key].itc += taxAmt;
        }
      });
    }

    const rawGSTR = ss.getSheetByName("GSTR_2A");
    if (rawGSTR) {
      rawGSTR.getDataRange().getDisplayValues().slice(1).forEach(r => {
        if (r[0] && liabilityMap[r[0]]) { liabilityMap[r[0]].itc += parseFloat(r[4] || 0); }
        else if (r[0]) { liabilityMap[r[0]] = { month: r[0], liability: 0, itc: parseFloat(r[4] || 0), net: 0, paid: 0 }; }
      });
    }

    const rawGSTTracker = ss.getSheetByName("GST_TRACKER");
    if (rawGSTTracker) {
      rawGSTTracker.getDataRange().getDisplayValues().slice(1).forEach(r => {
        const m = r[0];
        if (!liabilityMap[m]) liabilityMap[m] = { month: m, liability: 0, itc: 0, net: 0, paid: 0 };
        liabilityMap[m].paid = parseFloat(r[3]) || 0;
      });
    }

    // --- CHRONOLOGICAL ITC CARRY FORWARD ENGINE ---
    const sortedMonths = Object.keys(liabilityMap).sort(); // Sorts YYYY-MM oldest to newest
    let carryForwardITC = 0;

    sortedMonths.forEach(k => {
      // 1. Bring in balance from previous month
      liabilityMap[k].openingITC = carryForwardITC;

      // 2. Add this month's new purchases/inputs
      let totalAvailableITC = liabilityMap[k].itc + carryForwardITC;

      // 3. Subtract from Output Liability
      let net = liabilityMap[k].liability - totalAvailableITC;

      if (net < 0) {
        // We have excess ITC! 
        liabilityMap[k].net = 0;
        carryForwardITC = Math.abs(net); // Save it for next month
        liabilityMap[k].closingITC = carryForwardITC;
      } else {
        // We have to pay tax this month
        liabilityMap[k].net = net;
        carryForwardITC = 0; // ITC is drained
        liabilityMap[k].closingITC = 0;
      }
    });

    // Returns newest months first for the UI
    return JSON.stringify(Object.values(liabilityMap).sort((a, b) => b.month.localeCompare(a.month)));
  }
  return "{}";
}

// SITE-WISE P&L (WITH WORK ORDER BREAKDOWN)
// ==========================================
function getSitePnL(token,siteName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const invoices = ss.getSheetByName("Invoices").getDataRange().getDisplayValues();
  const payments = ss.getSheetByName("Payments").getDataRange().getDisplayValues();
  validateSession(token);
  
  let grandTotalRevenue = 0;
  let vendorCost = 0;
  let gstLiability = 0;
  const targetSite = String(siteName).trim().toLowerCase();
  
  let woBreakdown = {}; // 🚀 NEW: Tracks WO data

  // 1. Calculate Revenue & Group by Work Order
  invoices.slice(1).forEach(r => {
    if (r[38] === "TRUE") return;
    const isCancelled = (r[34] && r[34].includes("CANCELLED")) || (r[36] === "CANCELLED");
    const rSite = String(r[4] || "Main Site").trim().toLowerCase();
    
    if (!isCancelled && r[2] === "Tax Invoice" && rSite === targetSite) {
      const invTotal = parseFloat(r[29]) || 0;
      const tax = (parseFloat(r[25]) || 0) + (parseFloat(r[26]) || 0) + (parseFloat(r[27]) || 0);
      
      grandTotalRevenue += invTotal;
      gstLiability += tax;
      
      // Attribute revenue to Work Order
      let woNo = String(r[10] || "").trim().toUpperCase();
      if (!woNo) woNo = "Unlinked_Revenue"; // Catches general site invoices
      
      if (!woBreakdown[woNo]) woBreakdown[woNo] = { woNo: woNo, revenue: 0, cost: 0, profit: 0 };
      woBreakdown[woNo].revenue += invTotal;
    }
  });

  // 2. Calculate Expenses & Group by Work Order
  payments.slice(1).forEach(r => {
    if (r[11] === "TRUE") return;
    const type = String(r[3] || "").trim();
    const amount = parseFloat(r[5]) || 0;
    const rSite = String(r[7] || "Main Site").trim().toLowerCase();
    
    if (rSite === targetSite) {
      if (type === "Vendor Payment" || type === "Expense" || (type !== "Advance" && type !== "GST Payment" && amount < 0)) {
        const absAmount = Math.abs(amount);
        vendorCost += absAmount;
        
        // Attribute cost to Work Order (Column 14)
        let woNo = String(r[13] || "").trim().toUpperCase();
        if (!woNo) woNo = "UNLINKED_EXPENSES"; // Catches general site expenses
        
        if (!woBreakdown[woNo]) woBreakdown[woNo] = { woNo: woNo, revenue: 0, cost: 0, profit: 0 };
        woBreakdown[woNo].cost += absAmount;
      }
    }
  });

  // 3. Calculate WO Profit and Sort
  let woList = Object.keys(woBreakdown).map(k => {
      let w = woBreakdown[k];
      w.profit = w.revenue - w.cost;
      return w;
  });

  // Sort WO list (Largest Revenue first)
  woList.sort((a,b) => b.revenue - a.revenue);

  const netIncome = grandTotalRevenue - gstLiability;
  const tdsCalculated = grandTotalRevenue * 0.01;
  const profit = netIncome - vendorCost - tdsCalculated;

  return JSON.stringify({
    site: siteName,
    income: netIncome,
    vendor: vendorCost,
    gst: gstLiability,
    tds: tdsCalculated,
    profit: profit,
    workOrders: woList // 🚀 NEW: Sends the WO array to the frontend
  });
}

// PREMIUM: Rolling Backup System (Keeps only the last 7 days)
function createBackup() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const config = getConfig();
    const folderName = config.DRIVE_FOLDER || DEFAULT_DRIVE_FOLDER;
    let folder;
    // Find or create folder
    const folders = DriveApp.getFoldersByName(folderName);
    if (folders.hasNext()) folder = folders.next();
    else folder = DriveApp.createFolder(folderName);

    // 1. Create Today's Backup
    const timestamp = Utilities.formatDate(new Date(), "GMT+5:30", "yyyy-MM-dd_HH-mm");
    const backupName = `${ss.getName()}_Backup_${timestamp}`;
    DriveApp.getFileById(ss.getId()).makeCopy(backupName, folder);

    // 2. Storage Cleanup (Delete backups older than 7 days)
    const RETENTION_DAYS = 7;
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - RETENTION_DAYS);

    const files = folder.getFiles();
    let deletedCount = 0;
    while (files.hasNext()) {
      const file = files.next();
      const fileName = file.getName();
      // Only touch files that look like our backups
      if (fileName.includes("_Backup_")) {
        if (file.getDateCreated() < cutoffDate) {
          file.setTrashed(true);
          deletedCount++;
        }
      }
    }

    return JSON.stringify({
      status: "success",
      message: `Backup created successfully. Cleared ${deletedCount} old backups to save storage.`
    });
  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  }
}

function savePurchase(token, formStr) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    validateSession(token);
    const d = JSON.parse(formStr);
    hardValidate(d, 'Purchase');
    checkPeriodLocks(d.date);

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("Purchases");
    if (!sheet) {
      sheet = ss.insertSheet("Purchases");
    }

    if (sheet.getMaxColumns() < 14) {
      sheet.insertColumnsAfter(sheet.getMaxColumns(), 14 - sheet.getMaxColumns());
      sheet.getRange(1, 10).setValue("CostHead");
      sheet.getRange(1, 11).setValue("LastModified");
      sheet.getRange(1, 12).setValue("IsDeleted");
      sheet.getRange(1, 13).setValue("2A_Status");
      sheet.getRange(1, 14).setValue("WO_No"); // 🚀 NEW HEADER
    }
    const newLastModified = new Date().getTime().toString();
    const status2A = d.status2A || "Pending"; // Captures the new dropdown

    if (d.id) {
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (data[i][0] == d.id) {
          let currentLastModified = String(data[i][10] || "").trim();
          if (d.lastModified && currentLastModified && currentLastModified !== d.lastModified) {
            return JSON.stringify({ status: "error", message: "Concurrency Error: This purchase was modified by another user." });
          }

          // Update it in the "Edit" block AND the "New Record" block
        const row = [d.id, d.date, d.vendor, d.gstin, d.invNo, d.taxable, d.tax, d.total, d.desc, d.costHead, newLastModified, "", status2A, d.woNo || ""];
          sheet.getRange(i + 1, 1, 1, row.length).setValues([row]);
          logDetailedAudit("Purchases", "EDIT", d.id, data[i], d);

          try { manageJournalEntry("PURCHASE", d.id, d.date, []); } catch (e) { }
          break;
        }
      }
    } else {
      d.id = "PUR-" + Date.now();
      sheet.appendRow([d.id, d.date, d.vendor, d.gstin, d.invNo, d.taxable, d.tax, d.total, d.desc, d.costHead, newLastModified, "", status2A]);
      logDetailedAudit("Purchases", "CREATE", d.id, {}, d);
    }

    try {
      let pTaxable = parseFloat(d.taxable) || 0;
      let pTax = parseFloat(d.tax) || 0;
      let pTotal = parseFloat(d.total) || 0;
      // Auto-correct unbalanced inputs to prevent silent ledger corruption
      if (Math.abs((pTaxable + pTax) - pTotal) > 0.1) {
        if (pTaxable === 0 && pTotal > 0) pTaxable = pTotal - pTax;
        else pTotal = pTaxable + pTax;
      }

      manageJournalEntry("PURCHASE", d.id, d.date, [
        { acc: "Purchase / " + (d.costHead || "Expense"), dr: pTaxable, cr: 0 },
        { acc: "GST Input (ITC)", dr: pTax, cr: 0 },
        { acc: d.vendor + " (AP)", dr: 0, cr: pTotal }
      ]);
    } catch (e) { logSystemError("savePurchase_Journal", e.message); }

    SpreadsheetApp.flush(); // <--- FORCES GOOGLE TO WRITE DATA INSTANTLY
    Utilities.sleep(1500); // <-- ADDED HERE TOO
    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success" });
  } catch (e) {
    logSystemError("savePurchase", e.message);
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally { lock.releaseLock(); }
}

  const adminPin = String(config.ACCESS_PIN).trim();

  const providedPin = String(pin).trim();
  const providedHash = hashString(providedPin);

  // This checks if the saved PIN is plain-text OR encrypted
  if (providedPin === adminPin || providedHash === adminPin) return { valid: true, role: 'admin' };
  if ((providedPin === userPin || providedHash === userPin) && userPin !== "") return { valid: true, role: 'user' };

  return { valid: false, role: 'none' };
}

// BULK PAYMENT AUTO-ALLOCATION ENGINE
// ==========================================
function processBulkPayment(token, payloadStr) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    validateSession(token);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const invSheet = ss.getSheetByName("Invoices");
    const paySheet = ss.getSheetByName("Payments");

    const payload = JSON.parse(payloadStr);
    const clientName = payload.clientName;
    let remainingMoney = parseFloat(payload.amount);

    if (isNaN(remainingMoney) || remainingMoney <= 0) {
      return JSON.stringify({ status: "error", message: "Invalid payment amount." });
    }

    const payDateStr = payload.date || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
    const mode = payload.mode || "Bank Transfer";
    const receiptUrl = payload.receiptUrl || "";

    // 1. Get Payments & Map Headers Dynamically
    const payData = paySheet.getDataRange().getValues();
    const payHeaders = payData[0];
    const pRefIdx = payHeaders.indexOf("Ref") !== -1 ? payHeaders.indexOf("Ref") : 4;
    const pAmtIdx = payHeaders.indexOf("Amount") !== -1 ? payHeaders.indexOf("Amount") : 5;

    let payMap = {};
    for (let i = 1; i < payData.length; i++) {
      let ref = payData[i][pRefIdx];
      if (ref) {
        payMap[ref] = (payMap[ref] || 0) + (parseFloat(payData[i][pAmtIdx]) || 0);
      }
    }

    // 2. Get Invoices & Map Headers Dynamically (Failsafe)
    const invData = invSheet.getDataRange().getValues();
    const invHeaders = invData[0];

    const iNumIdx = invHeaders.indexOf("InvNum") !== -1 ? invHeaders.indexOf("InvNum") : 1;
    const iDateIdx = invHeaders.indexOf("Date") !== -1 ? invHeaders.indexOf("Date") : 2;
    const iTypeIdx = invHeaders.indexOf("Type") !== -1 ? invHeaders.indexOf("Type") : 3;
    const iStatusIdx = invHeaders.indexOf("Status") !== -1 ? invHeaders.indexOf("Status") : 4;
    const iClientIdx = invHeaders.indexOf("Company") !== -1 ? invHeaders.indexOf("Company") : (invHeaders.indexOf("ClientName") !== -1 ? invHeaders.indexOf("ClientName") : 5);
    const iSiteIdx = invHeaders.indexOf("SiteName") !== -1 ? invHeaders.indexOf("SiteName") : (invHeaders.indexOf("Site") !== -1 ? invHeaders.indexOf("Site") : 6);
    const iTotalIdx = invHeaders.indexOf("Total") !== -1 ? invHeaders.indexOf("Total") : (invHeaders.indexOf("GrandTotal") !== -1 ? invHeaders.indexOf("GrandTotal") : 8);

    let pendingInvoices = [];
    let totalDuesSelected = 0;

    for (let r = 1; r < invData.length; r++) {
      const row = invData[r];

      if (row[iTypeIdx] === "Tax Invoice" && row[iStatusIdx] !== "Cancelled" && row[iClientIdx] === clientName) {
        let inum = row[iNumIdx];
        let tot = parseFloat(row[iTotalIdx]) || 0;
        let due = tot - (payMap[inum] || 0);

        if (due > 0.05) {
          totalDuesSelected += due;
          pendingInvoices.push({ invNum: inum, due: due, client: row[iClientIdx], site: row[iSiteIdx], date: new Date(row[iDateIdx]).getTime() });
        }
      }
    }

    // 3. Security Check: Prevent Over-allocation
    if (remainingMoney > (totalDuesSelected + 1)) {
      return JSON.stringify({ status: "error", message: `Blocked! You entered ₹${remainingMoney.toLocaleString('en-IN')}, but ${clientName} only owes ₹${totalDuesSelected.toLocaleString('en-IN')}.` });
    }

    // 4. Sort by Date (Pay oldest first)
    pendingInvoices.sort((a, b) => a.date - b.date);

    let payRowsToAppend = [];
    let ts = new Date().getTime();

    // Find max PAY ID before the loop
    let maxPayBulk = 0;
    for (let i = 1; i < payData.length; i++) {
      let m = String(payData[i][0]).trim().match(/^PAY-(\d{1,3})$/);
      if (m) maxPayBulk = Math.max(maxPayBulk, parseInt(m[1], 10));
    }

    for (let i = 0; i < pendingInvoices.length; i++) {
      let item = pendingInvoices[i];
      if (remainingMoney <= 0.05) break;

      let payAmount = (remainingMoney >= item.due) ? item.due : remainingMoney;
      remainingMoney -= payAmount;

      // Auto-increment for each bulk payment row
      maxPayBulk++;
      let payId = "PAY-" + String(maxPayBulk).padStart(3, '0');

      // FIX: Forces 13 columns so the database aligns perfectly with the UI
      payRowsToAppend.push([payId, payDateStr, item.client, "Invoice", item.invNum, payAmount, mode, item.site, "Bulk Auto-Allocation", "", ts, "", receiptUrl]);
    }

    if (payRowsToAppend.length > 0) {
      paySheet.getRange(paySheet.getLastRow() + 1, 1, payRowsToAppend.length, payRowsToAppend[0].length).setValues(payRowsToAppend);

      // SECURITY COMPLIANCE: Push to immutable Audit Log and clear cache
      if (typeof logAction === 'function') logAction("BULK_PAYMENT_ALLOC", parseFloat(payload.amount));
      CacheService.getScriptCache().remove("appData_final_chunks");

      return JSON.stringify({ status: "success", message: `Successfully allocated ₹${parseFloat(payload.amount).toLocaleString('en-IN')} across ${payRowsToAppend.length} invoices.` });
    } else {
      return JSON.stringify({ status: "warning", message: "Client has no pending dues." });
    }

  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

function recordTDS(token, payloadStr) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    validateSession(token); // BACKEND SECURITY CHECK

    const d = JSON.parse(payloadStr);
    if (!d.invNum || !d.amount || parseFloat(d.amount) <= 0) {
      return JSON.stringify({ status: "error", message: "Invalid TDS amount or Invoice reference." });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const invSheet = ss.getSheetByName("Invoices");
    const invoices = invSheet.getDataRange().getDisplayValues();
    let clientName = "";
    let site = "";

    for (let i = 1; i < invoices.length; i++) {
      if (invoices[i][6] === d.invNum && invoices[i][38] !== "TRUE") {
        clientName = invoices[i][5];
        site = invoices[i][4] || "Main Site";
        break;
      }
    }

    if (!clientName) {
      return JSON.stringify({ status: "error", message: "Invoice not found or deleted." });
    }

    // SMART SEQUENTIAL ID FOR TDS
    const paySheet = ss.getSheetByName("Payments");
    const allPayData = paySheet.getDataRange().getValues();
    let maxTDS = 0;
    for (let i = 1; i < allPayData.length; i++) {
      let m = String(allPayData[i][0]).trim().match(/^TDS-(\d{1,3})$/);
      if (m) maxTDS = Math.max(maxTDS, parseInt(m[1], 10));
    }
    const tdsPayId = "TDS-" + String(maxTDS + 1).padStart(3, '0');

    const payData = {
      id: tdsPayId,
      clientName: clientName,
      site: site,
      date: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd"),
      amount: parseFloat(d.amount),
      mode: "TDS Adjustment",
      type: "Invoice",
      ref: d.invNum,
      head: "TDS Deduction",
      lastModified: new Date().getTime().toString()
    };

    // (The duplicate paySheet declaration was removed from here!)
    const newLastModified = new Date().getTime().toString();
    const row = [
      payData.id, payData.date, payData.clientName, payData.type, payData.ref,
      payData.amount, payData.mode, payData.site, payData.head, "", newLastModified, "", ""
    ];

    paySheet.appendRow(row);

    manageJournalEntry("TDS", payData.id, payData.date, [
      { acc: "TDS Receivable (Asset)", dr: payData.amount, cr: 0 },
      { acc: clientName + " (AR)", dr: 0, cr: payData.amount }
    ]);
    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success", message: `₹${payData.amount} TDS successfully recorded for ${d.invNum}` });
  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

function getFinancialStatements() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let jSheet = ss.getSheetByName("Journal");
    if (!jSheet) return JSON.stringify({ trialBalance: [], balanceSheet: { assets: [], liabilities: [], equity: [] } });
    const journal = jSheet.getDataRange().getValues();
    const tbMap = {};

    for (let i = 1; i < journal.length; i++) {
      const r = journal[i];
      if (r[8] === "TRUE" || r[8] === true) continue;

      let acc = r[5];
      if (acc === "IGST Payable") acc = "IGST Account"; // Rename for clarity
      const dr = parseFloat(r[6]) || 0;
      const cr = parseFloat(r[7]) || 0;

      if (!tbMap[acc]) tbMap[acc] = { account: acc, dr: 0, cr: 0, net: 0 };
      tbMap[acc].dr += dr;
      tbMap[acc].cr += cr;
    }

    const trialBalance = Object.keys(tbMap).map(k => {
      const net = tbMap[k].dr - tbMap[k].cr;
      tbMap[k].net = net;
      return tbMap[k];
    }).filter(a => Math.abs(a.net) > 0.01);
    let assets = [];
    let liabilities = [];
    let equity = [];
    let retainedEarnings = 0;
    trialBalance.forEach(acc => {
      const name = String(acc.account).toLowerCase();
      const netVal = acc.net;

      if (name.includes("revenue") || name.includes("sales")) {
        retainedEarnings += netVal;
      } else if (name.includes("expense") || name.includes("purchase")) {
        retainedEarnings += netVal;
      }
      else if (name.includes("(ar)") || name.includes("bank/cash") || name.includes("receivable") || name.includes("asset") || name.includes("itc") || (name.includes("igst") && netVal > 0)) {
        if (netVal > 0) assets.push({ name: acc.account, value: netVal });
        else liabilities.push({ name: acc.account, value: Math.abs(netVal) });
      } else if (name.includes("(ap)") || name.includes("payable") || name.includes("liability") || (name.includes("igst") && netVal < 0)) {
        if (netVal < 0) liabilities.push({ name: acc.account, value: Math.abs(netVal) });
        else assets.push({ name: acc.account, value: netVal });
      } else if (name.includes("capital") || name.includes("equity")) {
        equity.push({ name: acc.account, value: Math.abs(netVal) });
      } else {
        if (netVal > 0) assets.push({ name: acc.account, value: netVal });
        else liabilities.push({ name: acc.account, value: Math.abs(netVal) });
      }
    });
    equity.push({ name: "Retained Earnings (P&L)", value: Math.abs(retainedEarnings), isLoss: retainedEarnings > 0 });
    return JSON.stringify({ status: "success", trialBalance, balanceSheet: { assets, liabilities, equity } });
  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  }
}

function saveManualJournal(payloadStr) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    const d = JSON.parse(payloadStr);
    if (!d.date || !d.entries || d.entries.length < 2) {
      return JSON.stringify({ status: "error", message: "Invalid Journal Payload" });
    }

    let totalDr = 0;
    let totalCr = 0;
    d.entries.forEach(e => {
      totalDr += (parseFloat(e.dr) || 0);
      totalCr += (parseFloat(e.cr) || 0);
    });
    if (Math.abs(totalDr - totalCr) > 0.01) {
      return JSON.stringify({ status: "error", message: "Debits and Credits must balance!" });
    }

    const journalId = "MAN-" + Date.now();
    manageJournalEntry("MANUAL_JOURNAL", journalId, d.date, d.entries);
    logAction("MANUAL_JOURNAL", `Amount: ₹${totalDr}`);
    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success", message: "Manual Journal Posted Successfully." });
  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

function putCacheChunked(cache, key, stringValue, timeInSeconds) {
  const chunkSize = 90000;
  const chunks = Math.ceil(stringValue.length / chunkSize);

  // FIX: Removed the extra "_chunks" string so the master key matches exactly
  cache.put(key, chunks.toString(), timeInSeconds);

  for (let i = 0; i < chunks; i++) {
    cache.put(key + "_" + i, stringValue.substring(i * chunkSize, (i + 1) * chunkSize), timeInSeconds);
  }
}

function getCacheChunked(cache, key) {
  // FIX: Removed the extra "_chunks" string here too
  const chunks = cache.get(key);

  if (!chunks) return null;
  let result = "";
  for (let i = 0; i < parseInt(chunks); i++) {
    let chunk = cache.get(key + "_" + i);
    if (!chunk) return null; // If a piece expired, fail safely
    result += chunk;
  }
  return result;
}

// PREMIUM ERP FEATURE: Intelligence & Integrity Scanner
function runSystemHealthScan() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let issues = [];

    const invoices = ss.getSheetByName("Invoices").getDataRange().getValues();
    const payments = ss.getSheetByName("Payments").getDataRange().getValues();
    const journal = ss.getSheetByName("Journal") ? ss.getSheetByName("Journal").getDataRange().getValues() : [];
    const gstTracker = ss.getSheetByName("GST_TRACKER") ? ss.getSheetByName("GST_TRACKER").getDataRange().getValues() : [];

    // 1. Duplicate Invoice Scan
    let invSet = new Set();
    for (let i = 1; i < invoices.length; i++) {
      if (invoices[i][38] === "TRUE") continue;
      let num = String(invoices[i][6]).trim();
      if (num && invSet.has(num)) {
        issues.push(`CRITICAL: Duplicate Invoice Number detected in database: ${num}`);
      }
      invSet.add(num);
    }

    // 2. GST Compliance Pre-Close Scan (Are there unpaid GST invoices in locked/past months?)
    let filedMonths = new Set();
    for (let i = 1; i < gstTracker.length; i++) {
      if (gstTracker[i][2] === "Filed") filedMonths.add(gstTracker[i][0]);
    }

    let unpaidGstCount = 0;
    for (let i = 1; i < invoices.length; i++) {
      if (invoices[i][38] === "TRUE" || invoices[i][2] !== "Tax Invoice") continue;
      const isCancelled = (invoices[i][34] && String(invoices[i][34]).includes("CANCELLED")) || invoices[i][36] === "CANCELLED";
      if (!isCancelled && invoices[i][35] !== "GST_PAID") {
        const dateObj = new Date(invoices[i][7]);
        if (!isNaN(dateObj.getTime())) {
          const mKey = dateObj.getFullYear() + "-" + String(dateObj.getMonth() + 1).padStart(2, '0');
          if (filedMonths.has(mKey)) {
            unpaidGstCount++;
          }
        }
      }
    }
    if (unpaidGstCount > 0) {
      issues.push(`COMPLIANCE WARNING: You have ${unpaidGstCount} invoices marked as 'GST Unpaid' in months where GSTR-3B is already marked as Filed.`);
    }

    // 3. Unbalanced Manual Journal Scan
    let journalBatch = {};
    for (let i = 1; i < journal.length; i++) {
      if (journal[i][8] === "TRUE") continue;
      let jId = journal[i][0];
      if (!journalBatch[jId]) journalBatch[jId] = { dr: 0, cr: 0, date: journal[i][3] };
      journalBatch[jId].dr += parseFloat(journal[i][6]) || 0;
      journalBatch[jId].cr += parseFloat(journal[i][7]) || 0;
    }
    for (const [jId, totals] of Object.entries(journalBatch)) {
      if (Math.abs(totals.dr - totals.cr) > 0.1) {
        issues.push(`FINANCIAL INTEGRITY: Journal ID ${jId} is unbalanced! (Dr: ${totals.dr}, Cr: ${totals.cr})`);
      }
    }

    if (issues.length === 0) {
      return JSON.stringify({ status: "success", message: "System Health Excellent. No integrity or compliance issues found." });
    } else {
      return JSON.stringify({ status: "warning", message: issues.join("\n\n") });
    }
  } catch (e) {
    return JSON.stringify({ status: "error", message: "Scanner failed: " + e.toString() });
  }
}

// MONTH-END VALIDATION WIZARD
// ==========================================
function runMonthEndChecklist(monthStr) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let issues = [];
    let status = "READY";

    const invoices = ss.getSheetByName("Invoices").getDataRange().getValues();
    const payments = ss.getSheetByName("Payments").getDataRange().getValues();
    const journals = ss.getSheetByName("Journal") ? ss.getSheetByName("Journal").getDataRange().getValues() : [];

    let targetY = parseInt(monthStr.split('-')[0]);
    let targetM = parseInt(monthStr.split('-')[1]);

    let payMap = {};
    payments.forEach(p => {
      if (p[11] === "TRUE" || p[11] === true) return;
      if (p[4]) payMap[p[4]] = (payMap[p[4]] || 0) + (parseFloat(p[5]) || 0);
    });

    // 1. Check Open Invoices & Negative Anomalies
    let openInvoices = 0;
    for (let i = 1; i < invoices.length; i++) {
      if (invoices[i][38] === "TRUE" || invoices[i][38] === true) continue;
      const invDate = new Date(invoices[i][7]);
      if (isNaN(invDate.getTime()) || invDate.getFullYear() !== targetY || (invDate.getMonth() + 1) !== targetM) continue;

      if (invoices[i][2] === "Tax Invoice") {
        const total = parseFloat(invoices[i][29]) || 0;
        const paid = payMap[invoices[i][6]] || 0;
        const isCancelled = (invoices[i][34] && String(invoices[i][34]).includes("CANCELLED")) || invoices[i][36] === "CANCELLED";

        if (!isCancelled) {
          if (paid < total) openInvoices++;
          if (paid > (total + 1)) {
            issues.push(`ANOMALY: Invoice ${invoices[i][6]} is overpaid (Total: ${total}, Paid: ${paid}).`);
            status = "NOT_READY";
          }
        }
      }
    }
    if (openInvoices > 0) issues.push(`WARNING: There are ${openInvoices} unpaid/partially paid invoices for this month.`);

    // 2. Check Journal Imbalances for the month
    let journalBatch = {};
    for (let i = 1; i < journals.length; i++) {
      if (String(journals[i][8]).toUpperCase() === "TRUE") continue;
      const jDate = new Date(journals[i][3]);
      if (isNaN(jDate.getTime()) || jDate.getFullYear() !== targetY || (jDate.getMonth() + 1) !== targetM) continue;

      let jId = journals[i][0];
      if (!journalBatch[jId]) journalBatch[jId] = { dr: 0, cr: 0 };
      journalBatch[jId].dr += parseFloat(journals[i][6]) || 0;
      journalBatch[jId].cr += parseFloat(journals[i][7]) || 0;
    }

    for (const [jId, totals] of Object.entries(journalBatch)) {
      if (Math.abs(totals.dr - totals.cr) > 0.1) {
        issues.push(`CRITICAL: Journal ID ${jId} is unbalanced (Dr: ${totals.dr}, Cr: ${totals.cr}). Fix before closing.`);
        status = "NOT_READY";
      }
    }

    return JSON.stringify({ status: status, issues: issues });
  } catch (e) {
    return JSON.stringify({ status: "ERROR", issues: [e.toString()] });
  }
}

function getGSTReconciliationReport(monthStr) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let targetY = parseInt(monthStr.split('-')[0]);
    let targetM = parseInt(monthStr.split('-')[1]);

    let outputGST = 0;
    let itcAvailable = 0;
    let gstPaid = 0;

    const invoices = ss.getSheetByName("Invoices").getDataRange().getValues();
    for (let i = 1; i < invoices.length; i++) {
      if (invoices[i][38] === "TRUE" || invoices[i][38] === true) continue;
      if (invoices[i][2] !== "Tax Invoice") continue;

      const isCancelled = (invoices[i][34] && String(invoices[i][34]).includes("CANCELLED")) || invoices[i][36] === "CANCELLED";
      if (isCancelled) continue;

      const d = new Date(invoices[i][7]);
      if (!isNaN(d.getTime()) && d.getFullYear() === targetY && (d.getMonth() + 1) === targetM) {
        outputGST += (parseFloat(invoices[i][25]) || 0) + (parseFloat(invoices[i][26]) || 0) + (parseFloat(invoices[i][27]) || 0);
      }
    }

    const purchases = ss.getSheetByName("Purchases") ? ss.getSheetByName("Purchases").getDataRange().getValues() : [];
    for (let i = 1; i < purchases.length; i++) {
      if (purchases[i][11] === "TRUE" || purchases[i][11] === true) continue;
      const d = new Date(purchases[i][1]);
      if (!isNaN(d.getTime()) && d.getFullYear() === targetY && (d.getMonth() + 1) === targetM) {
        itcAvailable += parseFloat(purchases[i][6]) || 0;
      }
    }

    const payments = ss.getSheetByName("Payments").getDataRange().getValues();
    for (let i = 1; i < payments.length; i++) {
      if (payments[i][11] === "TRUE" || payments[i][11] === true) continue;
      if (payments[i][3] === "GST Payment") {
        const d = new Date(payments[i][1]);
        if (!isNaN(d.getTime()) && d.getFullYear() === targetY && (d.getMonth() + 1) === targetM) {
          gstPaid += Math.abs(parseFloat(payments[i][5]) || 0);
        }
      }
    }

    let netLiability = outputGST - itcAvailable;
    if (netLiability < 0) netLiability = 0; // Carry forward scenario

    // Tolerance of ₹1 for rounding differences
    let mismatchAlert = false;
    if (netLiability > 0 && Math.abs(netLiability - gstPaid) > 1) {
      mismatchAlert = true;
    }

    return JSON.stringify({
      OutputGST: outputGST,
      ITCAvailable: itcAvailable,
      NetGSTLiability: netLiability,
      GSTPaidToGovt: gstPaid,
      MismatchAlert: mismatchAlert,
      GSTOverdueIndicator: netLiability > 0 && gstPaid === 0
    });

  } catch (e) {
    return JSON.stringify({ error: e.toString() });
  }
}

function triggerMonthlyExecutiveReport() {
  // This runs on the 1st of the month, so we analyze the PREVIOUS month
  const today = new Date();
  let y = today.getFullYear();
  let m = today.getMonth(); // 0-indexed, so if today is March (2), previous month is Feb (2)
  if (m === 0) { m = 12; y = y - 1; } // Handle January -> December rollover

  const monthStr = y + "-" + String(m).padStart(2, '0');
  const monthName = new Date(y, m - 1).toLocaleString('default', { month: 'long' });

  generateAndEmailMonthlyReport(monthStr, monthName, y);
}

// PDF CRYPTOGRAPHIC VERIFICATION ENGINE
// ==========================================
function verifyInvoiceHash(invNum, date, clientName, totalAmount, providedHash) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(3000);
    // This MUST exactly match the salt you used in the saveInvoice function!
    const secretSalt = "SAIDURGA-SECURE-2026";

    // FIX: The Auto-Filler drops the trailing ".00", but the original PDF had it.
    // We calculate BOTH mathematical versions to ensure a 100% flawless read.
    const rawTotal = String(totalAmount).trim();
    const fixedTotal = (parseFloat(totalAmount) || 0).toFixed(2);

    // Version 1: Exactly as typed
    const rawData1 = `${String(invNum).trim()}|${String(date).trim()}|${String(clientName).trim()}|${rawTotal}|${secretSalt}`;
    // Version 2: With enforced .00 decimals (Matches the original PDF Generator)
    const rawData2 = `${String(invNum).trim()}|${String(date).trim()}|${String(clientName).trim()}|${fixedTotal}|${secretSalt}`;

    // Calculate both digital fingerprints
    const hash1 = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, rawData1).map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');
    const hash2 = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, rawData2).map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');

    const provided = String(providedHash).trim().toLowerCase();

    // If EITHER of the fingerprints match the PDF, the document is 100% Authentic!
    const isMatch = (hash1 === provided || hash2 === provided);

    return JSON.stringify({ status: "success", match: isMatch });
  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

// YEAR-END ARCHIVAL ENGINE (TIME-LOCKED)
// ==========================================
function executeYearEndRollover(cutoffDate) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);

    // --- SECURITY: PREVENT MID-YEAR EXECUTION ---
    const currentDate = new Date();
    const currentMonth = currentDate.getMonth(); // 0 is January, 3 is April

    // Only allow archival during April (month index 3) and May (month index 4)
    if (currentMonth !== 3 && currentMonth !== 4) {
      return JSON.stringify({
        status: "error",
        message: "Archival Blocked: The Year-End Rollover can only be executed during the financial year transition period (April - May) to protect active accounting data."
      });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // Helper to safely get or create Archive sheets
    function getOrCreateSheet(name, headers) {
      let sheet = ss.getSheetByName(name);
      if (!sheet) {
        sheet = ss.insertSheet(name);
        sheet.appendRow(headers);
      }
      return sheet;
    }

    const invSheet = ss.getSheetByName("Invoices");
    const paySheet = ss.getSheetByName("Payments");

    const invData = invSheet.getDataRange().getValues();
    const payData = paySheet.getDataRange().getValues();

    if (invData.length <= 1) return JSON.stringify({ status: "warning", message: "No invoices to archive." });

    const invHeaders = invData[0];
    const payHeaders = payData[0];

    const archInvSheet = getOrCreateSheet("Archive_Invoices", invHeaders);
    const archPaySheet = getOrCreateSheet("Archive_Payments", payHeaders);

    const inIdx = invHeaders.indexOf("InvNum");
    const dateIdx = invHeaders.indexOf("Date");
    const totIdx = invHeaders.indexOf("Total");
    const payRefIdx = payHeaders.indexOf("Ref");
    const payAmtIdx = payHeaders.indexOf("Amount");

    // 1. Calculate how much has been paid for every invoice
    let payMap = {};
    for (let i = 1; i < payData.length; i++) {
      let ref = payData[i][payRefIdx];
      if (ref) payMap[ref] = (payMap[ref] || 0) + (parseFloat(payData[i][payAmtIdx]) || 0);
    }

    let keepInvoices = [invHeaders];
    let archiveInvoices = [];
    let archivedInvNums = new Set();
    const cutoff = new Date(cutoffDate).getTime();

    // 2. Separate Invoices (Only archive if older than cutoff AND fully paid)
    for (let i = 1; i < invData.length; i++) {
      let row = invData[i];
      let invDate = new Date(row[dateIdx]).getTime();
      let invNum = row[inIdx];
      let due = (parseFloat(row[totIdx]) || 0) - (payMap[invNum] || 0);

      if (invDate <= cutoff && due <= 5) { // 5 rupee tolerance for decimal errors
        archiveInvoices.push(row);
        archivedInvNums.add(invNum);
      } else {
        keepInvoices.push(row);
      }
    }

    // 3. Separate Payments (Only archive payments tied to archived invoices)
    let keepPayments = [payHeaders];
    let archivePayments = [];
    for (let i = 1; i < payData.length; i++) {
      let row = payData[i];
      if (archivedInvNums.has(row[payRefIdx])) archivePayments.push(row);
      else keepPayments.push(row);
    }

    // 4. Write to Archives
    if (archiveInvoices.length > 0) archInvSheet.getRange(archInvSheet.getLastRow() + 1, 1, archiveInvoices.length, archiveInvoices[0].length).setValues(archiveInvoices);
    if (archivePayments.length > 0) archPaySheet.getRange(archPaySheet.getLastRow() + 1, 1, archivePayments.length, archivePayments[0].length).setValues(archivePayments);

    // 5. Update Active Sheets (Keeps your formatting safe)
    invSheet.clearContents();
    if (keepInvoices.length > 0) invSheet.getRange(1, 1, keepInvoices.length, keepInvoices[0].length).setValues(keepInvoices);

    paySheet.clearContents();
    if (keepPayments.length > 0) paySheet.getRange(1, 1, keepPayments.length, keepPayments[0].length).setValues(keepPayments);

    CacheService.getScriptCache().remove("appData_final_chunks");

    return JSON.stringify({ status: "success", message: `Successfully archived ${archiveInvoices.length} invoices and ${archivePayments.length} associated payments!` });

  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

// TWO-FACTOR AUTHENTICATION (OTP) ENGINE
// ==========================================
function requestLoginOTP(role, pin) {
  try {
    // RATE LIMITER: 5 attempts per 15 mins
    const cache = CacheService.getScriptCache();
    const rateKey = "OTP_LOCKOUT_" + role;
    const attempts = parseInt(cache.get(rateKey)) || 0;

    if (attempts >= 5) {
      return JSON.stringify({ valid: false, message: "Security Block: Too many attempts. Try again in 15 minutes." });
    }
    cache.put(rateKey, (attempts + 1).toString(), 900);

    const access = checkRoleAccess(pin);

    // 2. Block access if the PIN is wrong or role doesn't match
    if (!access.valid || access.role !== role) {
      return JSON.stringify({ valid: false, message: "Incorrect PIN for the selected role." });
    }

    // 3. Generate a secure 6-Digit OTP
    const otp = Math.floor(100000 + Math.random() * 900000).toString();

    // 4. Store OTP in Google Cache for 5 minutes (300 seconds)
    CacheService.getScriptCache().put("OTP_" + role, otp, 300);

    // 5. Get the Owner's Email (The account that owns the script)
    const ownerEmail = Session.getEffectiveUser().getEmail();

    // 6. Send the Secure Email
    const htmlBody = `
            <div style="font-family: Arial, sans-serif; padding: 25px; border: 1px solid #e2e8f0; border-radius: 12px; max-width: 450px; margin: auto; background: #ffffff;">
                <h2 style="color: #007aff; text-align: center; margin-top: 0;">SD Dynamics Security</h2>
                <p style="color: #475569; font-size: 15px;">A login attempt was just made for the <b>${role.toUpperCase()}</b> dashboard.</p>
                <p style="color: #475569; font-size: 15px;">Your secure One-Time Password is:</p>
                <h1 style="text-align: center; background: #f1f5f9; padding: 15px; letter-spacing: 8px; color: #1e293b; border-radius: 8px; border: 1px dashed #cbd5e1;">${otp}</h1>
                <p style="color: #94a3b8; font-size: 12px; text-align: center; margin-bottom: 0;">This code will expire automatically in 5 minutes. Do not share it with anyone.</p>
            </div>
        `;

    MailApp.sendEmail({
      to: ownerEmail,
      subject: "🔒 SD Dynamics - Login Verification Code",

    });

    return JSON.stringify({ valid: true, step: "otp_sent", email: ownerEmail });

  } catch (e) {
    return JSON.stringify({ valid: false, message: "Error generating OTP: " + e.message });
  }
}

function verifyLoginOTP(role, enteredOtp) {
  try {
    const cachedOtp = CacheService.getScriptCache().get("OTP_" + role);

    if (!cachedOtp) {
      return JSON.stringify({ valid: false, message: "OTP has expired. Please refresh the page and try again." });
    }

    if (cachedOtp === String(enteredOtp).trim()) {
      CacheService.getScriptCache().remove("OTP_LOCKOUT_" + role);
      CacheService.getScriptCache().remove("OTP_" + role); // Destroy OTP so it can't be reused
      const sessionToken = Utilities.getUuid();
      CacheService.getScriptCache().put("SESSION_" + sessionToken, role, 28800); // 8 Hours token
      return JSON.stringify({ valid: true, role: role, token: sessionToken });
    } else {
      return JSON.stringify({ valid: false, message: "Invalid OTP. Please check your email and try again." });
    }
  } catch (e) {
    return JSON.stringify({ valid: false, message: "Server Verification Error." });
  }
}

}

      const invNum = String(data[rowIndex][6]).trim();

      // Check payments
      const paySheet = ss.getSheetByName("Payments");
      if (paySheet) {
        const payData = paySheet.getDataRange().getValues();
        let totalPaid = 0;
        for (let p = 1; p < payData.length; p++) {
          if (String(payData[p][11]).trim().toUpperCase() !== "TRUE" && 
              String(payData[p][4]).trim() === invNum) {
            totalPaid += parseFloat(payData[p][5]) || 0;
          }
        }
        if (totalPaid > 0) {
          return JSON.stringify({ status: "error", message: `Cannot delete. ₹${totalPaid} already received. Delete payment first.` });
        }
      }

      // Perform soft delete using column names or safer references if possible
      const statusCol = 37;   // Consider using getLastColumn() + header mapping in future
      const deletedCol = 39;

      sheet.getRange(rowIndex + 1, statusCol).setValue("CANCELLED");
      sheet.getRange(rowIndex + 1, deletedCol).setValue(true);

      deleteJournalEntries(id);  // Assume this function is safe
      const linkedWO = String(data[rowIndex][10]).trim();
      if (linkedWO && typeof recalculateWorkOrderStatus === "function") {
        recalculateWorkOrderStatus(linkedWO);
      }
    } 
    else {
      // HARD BATCH DELETE for everything else
      const originalLength = data.length;
      const newData = data.filter((row, idx) => 
        idx === 0 || String(row[idCol]).trim() !== idStr
      );

      if (newData.length === originalLength) {
        throw new Error("Record not found");
      }

      // Better than clearContents() — preserves formatting
      sheet.getRange(1, 1, data.length, data[0].length).clearContent();
      sheet.getRange(1, 1, newData.length, newData[0].length).setValues(newData);

      // Journal & WorkOrder cleanup
      if (["Payments", "WorkOrders", "Purchases"].includes(type)) {
        deleteJournalEntries(id);
      }

      if (type === "WorkOrders") {
        const invSheet = ss.getSheetByName("Invoices");
        if (invSheet) {
          let invData = invSheet.getDataRange().getValues();
          let changed = false;
          for (let i = 1; i < invData.length; i++) {
            if (String(invData[i][10]).trim().toUpperCase() === idStr.toUpperCase()) {
              invData[i][10] = "";
              changed = true;
            }
          }
          if (changed) {
            invSheet.getRange(1, 1, invData.length, invData[0].length).setValues(invData);
          }
        }
      }
    }

    // Clear cache
    CacheService.getScriptCache().remove("appData_final_chunks");

    return JSON.stringify({ status: "success", message: "Record successfully removed/cancelled" });

  } catch (e) {
    console.error("deleteData error:", e);
    return JSON.stringify({ status: "error", message: e.message || e.toString() });
  } finally {
    lock.releaseLock();
  }
}

function savePO(token, formStr) {
  validateSession(token);
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    const d = JSON.parse(formStr);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("PurchaseOrders");

    // Updated Headers to include Subject, IGST, CGST, SGST
    if (!sheet) {
      sheet = ss.insertSheet("PurchaseOrders");
      sheet.appendRow(["PO Number", "Date", "Vendor", "Vendor Addr", "Site", "Ship Addr", "Status", "Tax Rate", "Subtotal", "Total GST", "Total", "Notes", "Terms", "Items JSON", "PDF Link", "Last Modified", "Subject", "IGST", "CGST", "SGST"]);
    }

    // --- SIGNATURE BASE64 CONVERTER ---
    const config = getConfig();
    let signatureUrl = config.SIGNATURE_URL || "";
    if (signatureUrl.includes("drive.google.com") && signatureUrl.includes("/d/")) {
      try {
        const id = signatureUrl.split('/d/')[1].split('/')[0];
        const imgBlob = DriveApp.getFileById(id).getBlob();
        signatureUrl = `data:${imgBlob.getContentType()};base64,${Utilities.base64Encode(imgBlob.getBytes())}`;
      } catch (e) { }
    }
    d.signatureUrl = signatureUrl;

    // --- SHA-256 DIGITAL FINGERPRINT ---
    const secretSalt = "SAIDURGA-SECURE-2026";
    const rawData = `${d.poNum}|${d.date}|${d.vendor}|${d.total}|${secretSalt}`;
    const hashBytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, rawData);
    d.digitalFingerprint = hashBytes.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');

    // Generate the PDF
    const pdfUrl = generatePO_PDF(d);

    const data = sheet.getDataRange().getValues();
    let foundRow = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(d.poNum).trim()) { foundRow = i + 1; break; }
    }

    const now = new Date().toISOString();

    // ADDED: d.subject, igst, cgst, and sgst to the end of the database row
    const row = [
      d.poNum, d.date, d.vendor, d.vendorAddr, d.site, d.shipAddr, d.status, d.taxRate,
      parseFloat(d.subtotal), parseFloat(d.gstAmt), parseFloat(d.total),
      d.notes, d.terms, d.items, pdfUrl, now,
      d.subject || "", parseFloat(d.igst) || 0, parseFloat(d.cgst) || 0, parseFloat(d.sgst) || 0
    ];

    if (foundRow > -1) sheet.getRange(foundRow, 1, 1, row.length).setValues([row]);
    else sheet.appendRow(row);

    SpreadsheetApp.flush();
    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: 'success', url: pdfUrl });
  } catch (e) {
    if (typeof logSystemError === "function") logSystemError("savePO", e.message);
    return JSON.stringify({ status: 'error', message: e.message });
  } finally { lock.releaseLock(); }
}

// TRADING, P&L, AND CAPITAL ACCOUNT ENGINE
// ==========================================
function getFinalAccountsData() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const config = getConfig();

    // 1. Fetch Manual Settings (Defaults to 0 if you haven't set them yet)
    const openingStock = parseFloat(config.OPENING_STOCK) || 0;
    const closingStock = parseFloat(config.CLOSING_STOCK) || 0;
    const openingCapital = parseFloat(config.OPENING_CAPITAL) || 0;

    // 2. Build Cost Head Map (Tells the system what is Direct vs Indirect)
    let costHeadMap = {};
    const chSheet = ss.getSheetByName("CostHeads");
    if (chSheet) {
      const chData = chSheet.getDataRange().getValues();
      for (let i = 1; i < chData.length; i++) {
        // Map the exact Name to its Category (Direct/Indirect)
        costHeadMap[String(chData[i][1]).trim()] = String(chData[i][2]).trim();
      }
    }

    // Data Containers for the PDF
    let sales = 0;
    let directExpenses = {};
    let indirectExpenses = {};
    let drawings = 0;

    // 3. Calculate True Sales (from Invoices, excluding cancelled/deleted)
    const invoices = ss.getSheetByName("Invoices").getDataRange().getValues();
    for (let i = 1; i < invoices.length; i++) {
      if (invoices[i][38] === "TRUE" || invoices[i][38] === true) continue;
      const isCancelled = (invoices[i][34] && String(invoices[i][34]).includes("CANCELLED")) || invoices[i][36] === "CANCELLED";

      if (!isCancelled && invoices[i][2] === "Tax Invoice") {
        sales += parseFloat(invoices[i][28]) || 0; // Adds Taxable Value only
      }
    }

    // 4. Calculate Purchases & Route to Direct/Indirect
    const purchases = ss.getSheetByName("Purchases") ? ss.getSheetByName("Purchases").getDataRange().getValues() : [];
    for (let i = 1; i < purchases.length; i++) {
      if (purchases[i][11] === "TRUE" || purchases[i][11] === true) continue;

      const head = String(purchases[i][9] || "Uncategorized").trim();
      const category = costHeadMap[head] === "Indirect" ? "Indirect" : "Direct"; // Purchases default to Direct
      const amount = parseFloat(purchases[i][5]) || 0; // Taxable Value

      if (category === "Direct") {
        directExpenses[head] = (directExpenses[head] || 0) + amount;
      } else {
        indirectExpenses[head] = (indirectExpenses[head] || 0) + amount;
      }
    }

    // 5. Calculate General Expenses & Owner Drawings (from Payments)
    const payments = ss.getSheetByName("Payments").getDataRange().getValues();
    for (let i = 1; i < payments.length; i++) {
      if (payments[i][11] === "TRUE" || payments[i][11] === true) continue;

      const type = String(payments[i][3] || "").trim();
      const head = String(payments[i][8] || "Uncategorized").trim();
      const amount = Math.abs(parseFloat(payments[i][5]) || 0);

      // MAGIC FIX: Catch Personal Drawings instantly
      if (head.toLowerCase().includes("drawing") || head.toLowerCase().includes("personal")) {
        drawings += amount;
        continue; // Skips adding to P&L, routes to Capital Account
      }

      if (type === "Expense" || type === "Vendor Payment") {
        const category = costHeadMap[head] === "Direct" ? "Direct" : "Indirect"; // General payments default to Indirect
        if (category === "Direct") {
          directExpenses[head] = (directExpenses[head] || 0) + amount;
        } else {
          indirectExpenses[head] = (indirectExpenses[head] || 0) + amount;
        }
      }
    }

    // 6. The Final Auditing Math
    let totalDirect = Object.values(directExpenses).reduce((a, b) => a + b, 0);
    let totalIndirect = Object.values(indirectExpenses).reduce((a, b) => a + b, 0);

    let grossProfit = (sales + closingStock) - (openingStock + totalDirect);
    let netProfit = grossProfit - totalIndirect;
    let closingCapital = openingCapital + netProfit - drawings;

    return JSON.stringify({
      status: "success",
      trading: {
        openingStock: openingStock,
        directExpenses: directExpenses,
        sales: sales,
        closingStock: closingStock,
        grossProfit: grossProfit
      },
      pnl: {
        grossProfit: grossProfit,
        indirectExpenses: indirectExpenses,
        netProfit: netProfit
      },
      capital: {
        openingCapital: openingCapital,
        netProfit: netProfit,
        drawings: drawings,
        closingCapital: closingCapital
      }
    });

  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  }
}

// TALLY-STYLE GSTR-2A AUTO-RECONCILIATION (UPGRADED)
// ==========================================
function runAutoReconciliation(token, targetMonth) {   // <--- ADD TOKEN HERE
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    validateSession(token); // <--- ADD THIS SECURITY LOCK

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const purSheet = ss.getSheetByName("Purchases");
    const g2aSheet = ss.getSheetByName("GSTR_2A");

    if (!purSheet || !g2aSheet) return JSON.stringify({ status: "error", message: "Missing required databases." });

    const purData = purSheet.getDataRange().getDisplayValues();
    const g2aData = g2aSheet.getDataRange().getDisplayValues();

    const cleanInv = (inv) => String(inv).replace(/[^a-zA-Z0-9]/g, '').toUpperCase().trim();

    // 1. Build the GSTR-2A map 
    let g2aMap = {};
    for (let i = 1; i < g2aData.length; i++) {
      let gMonth = String(g2aData[i][0]).trim();
      if (targetMonth && gMonth !== targetMonth) continue;

      let gGstin = String(g2aData[i][1]).trim().toUpperCase();
      let gInv = cleanInv(g2aData[i][3]);
      let gTaxRaw = String(g2aData[i][6] || "0").replace(/,/g, '');
      let gTax = parseFloat(gTaxRaw) || 0;

      if (gGstin && gInv) {
        let key = gGstin + "_" + gInv;
        // 🚀 FIX #2: Accumulate taxes for invoices with multiple rows!
        g2aMap[key] = (g2aMap[key] || 0) + gTax;
      }
    }

    let matchCount = 0;

    // 🚀 FIX #1 SETUP: Create an array to hold the updated column data
    // We initialize it with the existing Status column data
    let statusUpdates = [];

    // 2. Loop through ALL Purchases
    for (let i = 1; i < purData.length; i++) {
      let currentStatus = String(purData[i][12] || "").trim();
      let newStatus = currentStatus;

      if (String(purData[i][11]).trim().toUpperCase() !== "TRUE" && currentStatus !== "Matched") {
        let pGstin = String(purData[i][3]).trim().toUpperCase();
        let pInv = cleanInv(purData[i][4]);
        let pTaxRaw = String(purData[i][6] || "0").replace(/,/g, '');
        let pTax = parseFloat(pTaxRaw) || 0;

        let lookupKey = pGstin + "_" + pInv;

        // 3. THE MATCHING LOGIC
        if (g2aMap.hasOwnProperty(lookupKey)) {
          let diff = Math.abs(g2aMap[lookupKey] - pTax);
          if (diff <= 2.0) { // Allows a ₹2 variance
            newStatus = "Matched";
            matchCount++;
          }
        }
      }
      // Push the final status (either old or newly matched) to our array
      statusUpdates.push([newStatus]);
    }

    // 🚀 FIX #1 EXECUTION: Paste all status updates back to the sheet in exactly ONE fast call!
    if (statusUpdates.length > 0) {
      purSheet.getRange(2, 13, statusUpdates.length, 1).setValues(statusUpdates);
    }

    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success", message: `Auto-Reconciliation Complete! Successfully matched ${matchCount} purchases with the GST Portal.` });

  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

// BULK GSTR-2A UPLOADER (ENTERPRISE BATCH ENGINE)
// ==========================================
function saveBulkGSTR2A(token, jsonStr) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    validateSession(token); // 🔒 Security Check

    const records = JSON.parse(jsonStr);
    if (!records || records.length === 0) {
      return JSON.stringify({ status: "error", message: "No records to save." });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("GSTR_2A");

    // Auto-create sheet if it doesn't exist
    if (!sheet) {
      sheet = ss.insertSheet("GSTR_2A");
      sheet.appendRow(["Month", "GSTIN", "VendorName", "InvNo", "Date", "Taxable", "Tax"]);
    }

    // We assume all records belong to the target month selected in the UI
    const targetMonth = records[0].month;

    // 🚀 ENTERPRISE SPEED FIX: In-Memory Batch Processing
    const existingData = sheet.getDataRange().getValues();
    const finalData = [existingData[0]]; // Keep headers

    // 1. Instantly filter out the old data for this specific month in memory
    for (let i = 1; i < existingData.length; i++) {
      if (existingData[i][0] !== targetMonth) {
        finalData.push(existingData[i]); // Keep all other months untouched
      }
    }

    // 2. Attach the brand new rows from your CSV
    records.forEach(d => {
      finalData.push([d.month, d.gstin, d.name, d.invNo || "", d.date || "", d.taxable || 0, d.tax || 0]);
    });

    // 3. 🛟 DATA OVERFLOW NET: Check if the sheet has enough physical blank rows!
    const currentMaxRows = sheet.getMaxRows();
    if (finalData.length > currentMaxRows) {
      sheet.insertRowsAfter(currentMaxRows, (finalData.length - currentMaxRows) + 100);
    }

    // 4. Clear the sheet and paste everything back in 0.5 seconds!
    sheet.clearContents();
    sheet.getRange(1, 1, finalData.length, finalData[0].length).setValues(finalData);

    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success", message: `Successfully parsed and uploaded ${records.length} B2B invoices for ${targetMonth}.` });

  } catch (e) {
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

// VENDOR MANAGEMENT
// ==========================================
function saveVendor(token, formStr) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);

    // 1. Security check
    if (!token || token.length > 250) throw new Error("Security Block: Invalid token.");
    if (typeof validateSession === 'function') validateSession(token);

    const d = JSON.parse(formStr);
    if (!d.name || d.name.trim() === "") throw new Error("Vendor name is required.");

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("Vendors");

    // 2. Auto-create sheet if it doesn't exist yet
    if (!sheet) {
      sheet = ss.insertSheet("Vendors");
      sheet.appendRow(["ID", "Name", "Phone", "Email", "Address", "GSTIN", "LastModified", "IsDeleted"]);
    }

    const data = sheet.getDataRange().getDisplayValues();
    let foundRow = -1;

    // 3. Smart ID Generation (e.g., V-001, V-002)
    let id = d.id;
    if (!id || id === "") {
      let maxV = 0;
      for (let i = 1; i < data.length; i++) {
        let m = String(data[i][0]).trim().match(/^V-(\d+)$/);
        if (m) maxV = Math.max(maxV, parseInt(m[1], 10));
      }
      id = "V-" + String(maxV + 1).padStart(3, '0');
    } else {
      // Find existing record for updates
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(id)) {
          foundRow = i + 1; break;
        }
      }
    }

    const lastModified = new Date().getTime().toString();
    // FIXED ORDER: ID, Name, Phone, Email, Address, GSTIN
    const row = [id, d.name, d.phone, d.email, d.address, d.gstin, lastModified, ""];

    // 4. Write to Database
    if (foundRow > -1) {
      sheet.getRange(foundRow, 1, 1, row.length).setValues([row]);
    } else {
      sheet.appendRow(row);
    }

    // 5. Nuke the cache so the new vendor appears instantly!
    CacheService.getScriptCache().remove("appData_final_chunks");

    return JSON.stringify({ status: "success", message: "Vendor saved successfully" });

  } catch (e) {
    if (typeof logSystemError === "function") logSystemError("saveVendor", e.message);
    return JSON.stringify({ status: "error", message: e.toString() });
  } finally {
    lock.releaseLock();
  }
}

    const data = sheet.getDataRange().getValues();
    let foundRow = -1;

    for (let i = 1; i < data.length; i++) {
      // Invoice Number is in Column G (index 6), IsDeleted is Index 38
      if (String(data[i][6]).trim() === String(invNum).trim() && String(data[i][38]) !== "TRUE") {
        foundRow = i + 1;
        break;
      }
    }

    if (foundRow === -1) throw new Error("Could not find document " + invNum);

    // Cancel_Status is in Column AK (Column 37 in Google Sheets)
    sheet.getRange(foundRow, 37).setValue(lockMessage);

    CacheService.getScriptCache().remove("appData_final_chunks");
    return JSON.stringify({ status: "success" });
  } catch (e) {
    if (typeof logSystemError === "function") logSystemError("lockDocument", e.message);
    return JSON.stringify({ status: "error", message: e.message });
  } finally {
    lock.releaseLock();
  }
}
